\section*{Section A}

\begin{enumerate}

%── A1 ──────────────────────────────────────────────────────────────────────────
\item State the contrapositive of: ``If $x$ is prime and $x > 2$, then $x$ is odd.''

\ans{If $x$ is even, then $x$ is not prime or $x \le 2$.}
\method{The contrapositive of $P \land Q \implies R$ is $\neg R \implies \neg P \lor \neg Q$. $\neg(\text{odd}) = \text{even}$, so: ``If $x$ is even, then $x$ is not prime or $x \le 2$.''}
\inv{Is the original statement true for all positive integers $x$?}

%── A2 ──────────────────────────────────────────────────────────────────────────
\item What is the negation of: ``For all real numbers $x$, there exists a real number $y$ such that $x+y=0$.''

\ans{There exists a real number $x$ such that for all real numbers $y$, $x+y \neq 0$.}
\method{Negating $\forall x \, \exists y \, (x+y=0)$ yields $\exists x \, \forall y \, (x+y \neq 0)$.}
\inv{Is the original statement true or false in $\mathbb{R}$?}

%── A3 ──────────────────────────────────────────────────────────────────────────
\item True or false: ``If a conditional statement $P \implies Q$ is true, its converse $Q \implies P$ must also be true.''

\ans{False}
\method{Conditional statements and their converses are logically independent. For example, $x=2 \implies x^2=4$ is true, but $x^2=4 \implies x=2$ is false ($x=-2$).}
\inv{Which related statement is logically equivalent to $P \implies Q$?}

%── A4 ──────────────────────────────────────────────────────────────────────────
\item Which of the following is equivalent to $\neg(P \implies Q)$? (A) $\neg P \implies \neg Q$, (B) $P \land \neg Q$, (C) $\neg P \lor Q$.

\ans{(B) $P \land \neg Q$}
\method{$P \implies Q \equiv \neg P \lor Q$. Taking the negation: $\neg(\neg P \lor Q) \equiv P \land \neg Q$ by De Morgan's laws.}
\inv{Under what truth values of $P$ and $Q$ is $P \implies Q$ false?}

%── A5 ──────────────────────────────────────────────────────────────────────────
\item What single counterexample disproves: ``$n^2+n+17$ is prime for all non-negative integers $n$?''

\ans{$n=17$ (or $n=16$)}
\method{For $n=17$, $17^2 + 17 + 17 = 17(17+1+1) = 17 \times 19 = 323$, which is composite ($17 \times 19$).}
\inv{What is the smallest non-negative integer counterexample for $n^2+n+41$?}

%── A6 ──────────────────────────────────────────────────────────────────────────
\item If $A \implies B$ and $B \implies C$, what logical rule allows concluding $A \implies C$?

\ans{Hypothetical Syllogism (or Transitivity of Implication)}
\method{Hypothetical Syllogism states that $(A \implies B) \land (B \implies C) \implies (A \implies C)$.}
\inv{Does transitivity hold for bidirectional equivalence ($\iff$)?}

%── A7 ──────────────────────────────────────────────────────────────────────────
\item True or false: ``An assumption that leads to $0=1$ in a proof by contradiction proves the original premise false.''

\ans{True}
\method{Assuming $\neg P$ and deriving an impossible contradiction like $0=1$ proves $\neg P$ is false, which forces $P$ to be true.}
\inv{Why is deriving $0=0$ insufficient in a proof by contradiction?}

%── A8 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``If $x^2 > y^2$, then $x > y$.''

\ans{$x = -3, y = 2$}
\method{$(-3)^2 = 9 > 2^2 = 4$, but $-3 < 2$. The claim fails for negative numbers.}
\inv{For which real numbers $x,y$ is $x^2 > y^2 \implies x > y$ valid?}

%── A9 ──────────────────────────────────────────────────────────────────────────
\item What is the base case needed for an inductive proof that $3^n > 2^n + 10$ for $n \ge 3$?

\ans{$n = 3$}
\method{For $n=3$, $3^3 = 27$ and $2^3 + 10 = 8 + 10 = 18$. Since $27 > 18$, $n=3$ is the valid base case.}
\inv{Why does the inequality fail for $n=1$ and $n=2$?}

%── A10 ──────────────────────────────────────────────────────────────────────────
\item True or false: ``The negation of 'every student passed' is 'every student failed'.''

\ans{False}
\method{The negation of $\forall x \, P(x)$ (every student passed) is $\exists x \, \neg P(x)$ (at least one student did not pass), not $\forall x \, \neg P(x)$.}
\inv{What is the difference between negating a universal quantifier vs swapping it to an opposite universal?}

\end{enumerate}

\section*{Section B}

\begin{enumerate}

%── B1 ──────────────────────────────────────────────────────────────────────────
\item Prove by contrapositive: ``If $n^2$ is an odd integer, then $n$ is an odd integer.''

\ans{Proof by contrapositive}
\method{The contrapositive is: ``If $n$ is even, then $n^2$ is even.'' \\
Proof: If $n$ is even, $n = 2k$ for some integer $k$. Then $n^2 = (2k)^2 = 4k^2 = 2(2k^2)$. Since $2k^2 \in \mathbb{Z}$, $n^2$ is even. Since the contrapositive is true, the original statement is true.}
\inv{How does this compare to a direct proof of the same claim?}

%── B2 ──────────────────────────────────────────────────────────────────────────
\item Find all integer counterexamples in $-10 \le x \le 10$ to: ``If $x^2 > 4$, then $x > 2$.''

\ans{$x \in \{-10, -9, -8, -7, -6, -5, -4, -3\}$}
\method{$x^2 > 4 \iff x > 2 \text{ or } x < -2$. The integers in $[-10, 10]$ satisfying $x < -2$ are $-10, -9, -8, -7, -6, -5, -4, -3$. For all these values, $x^2 > 4$ is true, but $x > 2$ is false.}
\inv{How many integer counterexamples exist in total?}

%── B3 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction: ``$\sqrt{3}$ is an irrational number.''

\ans{Proof by contradiction}
\method{Suppose toward a contradiction that $\sqrt{3} = p/q$ where $p,q \in \mathbb{Z}^+$ and $\gcd(p,q) = 1$. Squaring gives $3 = p^2/q^2 \implies p^2 = 3q^2$. Thus $3 \mid p^2 \implies 3 \mid p$, so $p = 3k$. Then $(3k)^2 = 3q^2 \implies 9k^2 = 3q^2 \implies q^2 = 3k^2 \implies 3 \mid q$. Thus $3 \mid \gcd(p,q)$, contradicting $\gcd(p,q) = 1$.}
\inv{Does this proof extend to $\sqrt{p}$ for any prime $p$?}

%── B4 ──────────────────────────────────────────────────────────────────────────
\item Classify the condition ``$x=2$'' as necessary, sufficient, both, or neither for ``$x^2 - 5x + 6 = 0$''.

\ans{Sufficient but not necessary}
\method{Sufficient: If $x=2$, $2^2 - 5(2) + 6 = 0$, so $x^2-5x+6=0$ holds. Not necessary: $x=3$ also satisfies $x^2-5x+6=0$, so $x=2$ is not required.}
\inv{Which set of values is necessary and sufficient for $x^2-5x+6=0$?}

%── B5 ──────────────────────────────────────────────────────────────────────────
\item Classify the condition ``$x^2 - 5x + 6 = 0$'' as necessary, sufficient, both, or neither for ``$x=2$''.

\ans{Necessary but not sufficient}
\method{Necessary: For $x=2$ to hold, $x^2-5x+6=0$ must hold. Not sufficient: $x^2-5x+6=0$ allows $x=3$, so it does not guarantee $x=2$.}
\inv{How does swapping hypothesis and conclusion swap necessary and sufficient roles?}

%── B6 ──────────────────────────────────────────────────────────────────────────
\item Prove by induction: $\sum_{i=1}^n 2^i = 2^{n+1} - 2$ for all positive integers $n$.

\ans{Proof by induction}
\method{Base case $n=1$: $\sum_{i=1}^1 2^i = 2$; $2^{1+1}-2 = 4-2 = 2$, true. \\
Inductive step: Assume $\sum_{i=1}^k 2^i = 2^{k+1}-2$. Then $\sum_{i=1}^{k+1} 2^i = (2^{k+1}-2) + 2^{k+1} = 2 \cdot 2^{k+1} - 2 = 2^{k+2} - 2$. By induction, the formula holds for all $n \ge 1$.}
\inv{What is the sum formula for a finite geometric series $\sum_{i=0}^{n-1} r^i$?}

%── B7 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``If $a,b$ are irrational, then $a^b$ is irrational.''

\ans{$a = \sqrt{2}^{\sqrt{2}}, b = \sqrt{2}$}
\method{If $\sqrt{2}^{\sqrt{2}}$ is rational, it is a counterexample ($a=\sqrt{2}, b=\sqrt{2}$ both irrational). If $\sqrt{2}^{\sqrt{2}}$ is irrational, take $a = \sqrt{2}^{\sqrt{2}}$ and $b = \sqrt{2}$, both irrational. Then $a^b = (\sqrt{2}^{\sqrt{2}})^{\sqrt{2}} = \sqrt{2}^{\sqrt{2} \times \sqrt{2}} = \sqrt{2}^2 = 2 \in \mathbb{Q}$, a rational number.}
\inv{This non-constructive proof establishes the existence of an irrational power of an irrational being rational without knowing which case is true!}

%── B8 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction: ``If $n$ is an integer and $n^3+5$ is odd, then $n$ is even.''

\ans{Proof by contradiction}
\method{Assume toward a contradiction that $n$ is odd, so $n = 2k+1$. Then $n^3+5 = (2k+1)^3+5 = 8k^3+12k^2+6k+1+5 = 2(4k^3+6k^2+3k+3)$, which is even. This contradicts $n^3+5$ being odd. Thus $n$ must be even.}
\inv{Could this also be proved directly by showing $n^3+5 \text{ odd} \implies n^3 \text{ even} \implies n \text{ even}$?}

%── B9 ──────────────────────────────────────────────────────────────────────────
\item State the negation of: ``For every $\varepsilon > 0$, there exists $\delta > 0$ such that if $|x-a| < \delta$, then $|f(x)-L| < \varepsilon$.''

\ans{There exists $\varepsilon > 0$ such that for all $\delta > 0$, there exists $x$ with $|x-a| < \delta$ and $|f(x)-L| \ge \varepsilon$.}
\method{Negating $\forall \varepsilon > 0 \, \exists \delta > 0 \, (P \implies Q)$ yields $\exists \varepsilon > 0 \, \forall \delta > 0 \, (P \land \neg Q)$. Here $P$ is $|x-a| < \delta$ and $\neg Q$ is $|f(x)-L| \ge \varepsilon$.}
\inv{This is the formal definition that $\lim_{x\to a} f(x) \neq L$ in analysis.}

%── B10 ──────────────────────────────────────────────────────────────────────────
\item Prove by induction that $7^n - 1$ is divisible by $6$ for all positive integers $n$.

\ans{Proof by induction}
\method{Base case $n=1$: $7^1 - 1 = 6$, divisible by $6$. \\
Inductive step: Assume $7^k - 1 = 6m$. Then $7^{k+1} - 1 = 7 \cdot 7^k - 1 = 7(6m+1) - 1 = 42m + 6 = 6(7m+1)$, which is divisible by $6$. By induction, $6 \mid (7^n-1)$ for all $n \ge 1$.}
\inv{How can this be proved using modular arithmetic ($7 \equiv 1 \pmod 6$)?}

\end{enumerate}

\section*{Section C}

\begin{enumerate}

%── C1 ──────────────────────────────────────────────────────────────────────────
\item For real numbers $x,y$, consider: \textit{\small(after TMUA 2022 Paper 2 Q10)}\\
I. $\forall x \, \exists y \; (x+y = 0)$\\
II. $\exists y \, \forall x \; (x+y = 0)$\\
III. $\forall x \, \forall y \; (x+y = 0)$\\
Which of I, II, III are true?

\ans{B) I only}
\method{I: For any given $x$, choosing $y = -x$ gives $x + (-x) = 0$, so I is true. \\
II: Requires a single fixed $y$ such that $x+y=0$ for ALL $x$. If $x=0 \implies y=0$, but then $1+0 \neq 0$, so II is false. \\
III: False (e.g. $x=1, y=1 \implies 1+1=2 \neq 0$). Thus I only is true.}
\inv{Why does order of quantifiers ($\forall x \exists y$ vs $\exists y \forall x$) change the meaning completely?}

%── C2 ──────────────────────────────────────────────────────────────────────────
\item Determine the complete set of real values $k$ for which the statement: ``For all real $x$, if $x > k$, then $x^2 > k$'' is true. \textit{\small(after TMUA 2022 Paper 2 Q9)}

\ans{A) $k \ge 1$ or $k \le 0$}
\method{Case $k \le 0$: If $x > k$, then $x^2 \ge 0 \ge k$. If $k=0$, $x > 0 \implies x^2 > 0 = k$. True for $k \le 0$. \\
Case $0 < k < 1$: Take $x \in (k, 1)$. Then $x > k$, but $x^2 < x < 1$. Specifically for $x = (k+1)/2$, $x > k$ holds, but $x^2 < k$ holds, so statement fails. \\
Case $k \ge 1$: If $x > k \ge 1$, then $x^2 > x > k$, true. \\
Thus the statement holds for $k \le 0$ or $k \ge 1$.}
\inv{Verify $k=0.5$: Take $x=0.6 > 0.5$, then $x^2 = 0.36 < 0.5$, counterexample!}

%── C3 ──────────────────────────────────────────────────────────────────────────
\item Consider positive integers $a,b$. Statement $P$: ``$a+b$ is even.'' Statement $Q$: ``$a^2 + b^2$ is even.'' Which of the following is true?

\ans{C) $P$ is necessary and sufficient for $Q$.}
\method{$a+b$ is even $\iff a,b$ have the same parity $\iff a^2, b^2$ have the same parity $\iff a^2+b^2$ is even. Thus $P \iff Q$, meaning $P$ is necessary and sufficient for $Q$.}
\inv{What if the statement was $a+b$ is odd vs $a^2+b^2$ is odd?}

%── C4 ──────────────────────────────────────────────────────────────────────────
\item A student attempts to prove: ``There are no integer solutions to $x^2 + y^2 = 7$.''

\ans{A) The proof is completely valid and correct.}
\method{Squares modulo $4$ are $0^2 \equiv 0$, $1^2 \equiv 1$, $2^2 \equiv 0$, $3^2 \equiv 1$. Thus $x^2 \pmod 4 \in \{0,1\}$ and $y^2 \pmod 4 \in \{0,1\}$. Their sum $(x^2+y^2) \pmod 4 \in \{0, 1, 2\}$. But $7 = 4(1) + 3 \equiv 3 \pmod 4$. Since $3 \notin \{0,1,2\}$, no integer solutions exist. The proof is sound.}
\inv{Which other integers cannot be written as the sum of two squares?}

%── C5 ──────────────────────────────────────────────────────────────────────────
\item For a real number $x$, let $P$: ``$x^2 - 4 = 0$'' and $Q$: ``$x = 2$''. Which of the following statements is true?

\ans{B) $Q \implies P$ is true, but $P \implies Q$ is false.}
\method{$Q \implies P$: If $x=2$, $2^2-4 = 0$, true. $P \implies Q$: $x^2-4=0 \implies x=2 \text{ or } x=-2$, so $x=-2$ is a counterexample to $P \implies Q$. Thus $Q \implies P$ is true, but $P \implies Q$ is false.}
\inv{Is $x=2$ necessary or sufficient for $x^2-4=0$?}

%── C6 ──────────────────────────────────────────────────────────────────────────
\item Consider the claim: ``For every positive integer $n$, $n^3 + 2n$ is divisible by $3$.'' Which proof technique is most direct?

\ans{A) Factorise $n^3+2n = n(n^2+2) = n(n^2-1+3) = n(n-1)(n+1) + 3n$, where $(n-1)n(n+1)$ is divisible by 3.}
\method{$n^3+2n = n^3-n+3n = (n-1)n(n+1) + 3n$. The product of 3 consecutive integers $(n-1)n(n+1)$ is divisible by 3, and $3n$ is divisible by 3, so their sum is divisible by 3 directly.}
\inv{Can this also be proved by testing $n \equiv 0, 1, 2 \pmod 3$?}

%── C7 ──────────────────────────────────────────────────────────────────────────
\item What is the negation of ``All prime numbers are odd or equal to 2''?

\ans{B) There exists a prime number that is even and not equal to 2.}
\method{Negating $\forall p, (P(p) \lor Q(p))$ gives $\exists p, (\neg P(p) \land \neg Q(p))$. $\neg(\text{odd}) = \text{even}$, and $\neg(p=2) = p \neq 2$. Thus: ``There exists a prime number that is even and not equal to 2.''}
\inv{Why does De Morgan's law switch 'or' to 'and' under negation?}

%── C8 ──────────────────────────────────────────────────────────────────────────
\item Let $S$ be a set of $n$ distinct positive integers. Consider the statement: ``If $n \ge 5$, then $S$ contains two numbers whose difference is a multiple of $4$.'' Is this statement true or false? \textit{\small(after SMC 2017 Q18)}

\ans{A) True, by the Pigeonhole Principle (4 possible remainders mod 4).}
\method{When an integer is divided by $4$, there are $4$ possible remainders: $0, 1, 2, 3$. By the Pigeonhole Principle, if $n \ge 5$ integers are selected, at least two must have the same remainder modulo $4$. If $a \equiv b \pmod 4$, then $a - b \equiv 0 \pmod 4$, meaning $a-b$ is a multiple of $4$.}
\inv{How many integers must be chosen to guarantee two whose difference is a multiple of $k$?}

\end{enumerate}

\section*{Section D}

\begin{enumerate}

%── D1 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction: ``There do not exist integers $x,y$ such that $x^2 - 5y^2 = 3$.'' \textit{\small(after BMO1 2015 Q1)}

\ans{Proof by contradiction}
\method{Assume toward a contradiction that there exist integers $x,y$ such that $x^2 - 5y^2 = 3$. Taking this equation modulo $5$, we get $x^2 \equiv 3 \pmod 5$. The squares modulo $5$ are $0^2 \equiv 0, 1^2 \equiv 1, 2^2 \equiv 4, 3^2 \equiv 4, 4^2 \equiv 1 \pmod 5$. Thus $x^2 \pmod 5 \in \{0, 1, 4\}$. Since $3 \notin \{0, 1, 4\}$, $x^2 \equiv 3 \pmod 5$ has no integer solutions. Contradiction.}
\inv{What are the quadratic residues modulo $5$?}

%── D2 ──────────────────────────────────────────────────────────────────────────
\item A student claims: ``If $a,b,c$ are positive integers such that $a \mid bc$, then $a \mid b$ or $a \mid c$.'' \textit{\small(after TMUA 2019 Paper 2 Q7)}

\ans{(a) $a=6, b=2, c=3$ ($6 \mid 6$ but $6 \nmid 2$ and $6 \nmid 3$). (b) $\gcd(a,b)=1$ (or $a$ is prime); Euclid's Lemma.}
\method{(a) Take $a=6, b=2, c=3$. $bc = 6$, so $6 \mid 6$ is true. However, $6 \nmid 2$ and $6 \nmid 3$, disproving the claim. \\
(b) The minimal condition is $\gcd(a,b)=1$ (or requiring $a$ to be prime). The result is Euclid's Lemma: If $a \mid bc$ and $\gcd(a,b)=1$, then $a \mid c$.}
\inv{Why does Euclid's Lemma require coprime factors or prime divisors?}

%── D3 ──────────────────────────────────────────────────────────────────────────
\item Prove by mathematical induction that for all positive integers $n$, $\sum_{i=1}^n i \cdot i! = (n+1)! - 1$.

\ans{Proof by induction}
\method{Base case $n=1$: $\sum_{i=1}^1 i \cdot i! = 1 \cdot 1! = 1$; $(1+1)! - 1 = 2! - 1 = 1$, true. \\
Inductive step: Assume $\sum_{i=1}^k i \cdot i! = (k+1)! - 1$. Then $\sum_{i=1}^{k+1} i \cdot i! = [(k+1)! - 1] + (k+1) \cdot (k+1)! = (k+1)! [1 + (k+1)] - 1 = (k+1)! (k+2) - 1 = (k+2)! - 1 = ((k+1)+1)! - 1$. By induction, the identity holds for all $n \ge 1$.}
\inv{How does this identity relate to telescopic sums using $i \cdot i! = (i+1)! - i!$?}

%── D4 ──────────────────────────────────────────────────────────────────────────
\item Consider three statements about a real function $f: \mathbb{R} \to \mathbb{R}$: \textit{\small(after TMUA 2023 Paper 2 Q12)}\\
I. If $f(x) > 0$ for all $x > 0$, then $f(1) > 0$.\\
II. If $f(1) > 0$, then $f(x) > 0$ for all $x > 0$.\\
III. If $f(1) \le 0$, then $f(x) \le 0$ for some $x > 0$.

\ans{I and III are logically equivalent to each other.}
\method{Statement I has the form $P \implies Q$, where $P$ is ``$f(x)>0$ for all $x>0$'' and $Q$ is ``$f(1)>0$''. Statement III is the contrapositive of I: $\neg Q$ is ``$f(1) \le 0$'', and $\neg P$ is ``$\exists x > 0, f(x) \le 0$''. Since a statement and its contrapositive are logically equivalent, I $\iff$ III. Statement II is the converse of I, which is not logically equivalent.}
\inv{Why does a statement always have the exact same truth value as its contrapositive?}

%── D5 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction that there are infinitely many prime numbers (Euclid's Theorem).

\ans{Proof by contradiction}
\method{Assume toward a contradiction that there are only finitely many prime numbers, $p_1, p_2, \dots, p_k$. Consider the integer $N = p_1 p_2 \dots p_k + 1$. Since $N > 1$, $N$ must have at least one prime factor $p$. Since $p_1, \dots, p_k$ are all the primes that exist, $p$ must be one of them, so $p \mid N$. But $p \mid (p_1 p_2 \dots p_k)$, so $p \mid (N - p_1 p_2 \dots p_k) \implies p \mid 1$. But no prime divides $1$, contradiction! Thus, infinitely many primes exist.}
\inv{Does Euclid's construction $N = p_1 p_2 \dots p_k + 1$ always yield a prime number itself?}

\end{enumerate}
