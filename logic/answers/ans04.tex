\section*{Section A}

\begin{enumerate}

%── A1 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``Every prime number is odd.''

\ans{$2$}
\method{$2$ is a prime number because its only positive divisors are $1$ and $2$, but $2$ is even. A single counterexample disproves the universal claim.}
\inv{Are there any other even prime numbers?}

%── A2 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``$n^2\geq n$ for every real number $n$.''

\ans{$n = 0.5$ (or any $n \in (0, 1)$)}
\method{For $n = 0.5 = 1/2$, we have $n^2 = 0.25$. Since $0.25 < 0.5$, $n^2 < n$, disproving the claim.}
\inv{For which interval of real numbers $n$ is $n^2 < n$ true?}

%── A3 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``If $n$ is prime, then $n+2$ is prime.''

\ans{$n = 7$ (or $n=23$)}
\method{$7$ is prime, but $7+2 = 9 = 3^2$, which is composite.}
\inv{Primes $p$ where $p+2$ is also prime are called twin primes. What is the smallest pair of twin primes?}

%── A4 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``Every multiple of $4$ is a multiple of $8$.''

\ans{$4$ (or $12, 20, 28, \dots$)}
\method{$4$ is a multiple of $4$ ($4 = 4 \times 1$), but $4$ is not a multiple of $8$ because $4/8 = 0.5 \notin \mathbb{Z}$.}
\inv{Is every multiple of $8$ a multiple of $4$?}

%── A5 ──────────────────────────────────────────────────────────────────────────
\item To prove ``there is no largest even integer'' by contradiction, what should you assume first?

\ans{Assume that there exists a largest even integer (call it $E$).}
\method{A proof by contradiction begins by assuming the exact negation of the statement to be proved.}
\inv{How do you then construct an even integer strictly larger than $E$?}

%── A6 ──────────────────────────────────────────────────────────────────────────
\item True or false: ``If assuming `not $P$' leads to something impossible, $P$ must be true.''

\ans{True}
\method{This is the fundamental principle of proof by contradiction (\textit{reductio ad absurdum}). Since a statement cannot be both true and false, establishing that $\neg P$ leads to a contradiction forces $P$ to be true.}
\inv{How does proof by contradiction differ from proof by contrapositive?}

%── A7 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``$\sqrt{a+b}=\sqrt a+\sqrt b$ for all $a,b\geq0$.''

\ans{$a=1, b=1$}
\method{For $a=1, b=1$, $\sqrt{a+b} = \sqrt{2}$, whereas $\sqrt{a} + \sqrt{b} = 1 + 1 = 2$. Since $\sqrt{2} \neq 2$, the identity fails.}
\inv{Under what condition on $a$ and $b$ does $\sqrt{a+b} = \sqrt{a} + \sqrt{b}$ hold?}

%── A8 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``$(x+y)^2=x^2+y^2$ for all real $x,y$.''

\ans{$x=1, y=1$}
\method{For $x=1, y=1$, $(x+y)^2 = 2^2 = 4$, but $x^2 + y^2 = 1^2 + 1^2 = 2$. Since $4 \neq 2$, the claim is false.}
\inv{Expand $(x+y)^2$ to state the exact condition required for $(x+y)^2 = x^2+y^2$.}

%── A9 ──────────────────────────────────────────────────────────────────────────
\item True or false: ``A single counterexample is enough to disprove a `for all' claim.''

\ans{True}
\method{The negation of $\forall x, P(x)$ is $\exists x, \neg P(x)$. Finding one single element $x$ where $P(x)$ is false satisfies the negation, disproving the universal claim.}
\inv{Is a single example enough to prove a `for all' claim?}

%── A10 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``$n!+1$ is always prime.''

\ans{$n=4$ (or $n=5$)}
\method{For $n=4$, $4! + 1 = 24 + 1 = 25 = 5^2$, which is composite. (For $n=5$, $5!+1 = 121 = 11^2$).}
\inv{Is $n!+1$ ever prime for $n > 3$?}

\end{enumerate}

\section*{Section B}

\begin{enumerate}

%── B1 ──────────────────────────────────────────────────────────────────────────
\item Five envelopes contain the numbers $6,9,14,21,25$ respectively. Each is sealed with a wax stamp showing either a star or a circle: the envelopes containing $6,14,25$ are star-sealed, and the envelopes containing $9,21$ are circle-sealed. Claim: ``Every star-sealed envelope contains an even number.'' Which envelope's contents provide a counterexample?

\ans{$25$}
\method{The star-sealed envelopes contain $6, 14, 25$. $6$ and $14$ are even, but $25$ is odd. The envelope containing $25$ is star-sealed yet contains an odd number, disproving the claim.}
\inv{How does this relate to Wason's Selection Task in logic?}

%── B2 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``If $a$ and $b$ are both irrational, then $a+b$ is irrational.''

\ans{$a = \sqrt{2}, b = -\sqrt{2}$}
\method{$\sqrt{2}$ and $-\sqrt{2}$ are both irrational numbers, but their sum is $\sqrt{2} + (-\sqrt{2}) = 0 = 0/1 \in \mathbb{Q}$, which is rational.}
\inv{Can you construct a counterexample where both $a$ and $b$ are positive?}

%── B3 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``If $a$ and $b$ are both irrational, then $ab$ is irrational.''

\ans{$a = \sqrt{2}, b = \sqrt{2}$}
\method{$\sqrt{2}$ is irrational, but $ab = \sqrt{2} \times \sqrt{2} = 2 = 2/1 \in \mathbb{Q}$, which is rational.}
\inv{Find a counterexample where $a \neq b$.}

%── B4 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction: ``There is no largest even integer.''

\ans{Proof by contradiction}
\method{Assume toward a contradiction that there exists a largest even integer, $E$. By definition of even integers, $E = 2k$ for some integer $k$. Consider $E' = E + 2 = 2(k+1)$. Since $k+1 \in \mathbb{Z}$, $E'$ is an even integer. Furthermore, $E' > E$. This contradicts the assumption that $E$ was the largest even integer. Thus, no largest even integer exists.}
\inv{Why must we add $2$ rather than $1$ to maintain even parity?}

%── B5 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample (using $n=0$) to: ``For every integer $n$, $n/n=1$.''

\ans{$n=0$}
\method{For $n=0$, $0/0$ is undefined in standard arithmetic. Since $0/0 \neq 1$, $n=0$ is a counterexample.}
\inv{What happens to the expression $x/x$ as $x \to 0$ in calculus?}

%── B6 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample (using $n=1$) to: ``Every positive integer has at least one prime factor.''

\ans{$n=1$}
\method{The prime factorisation of $1$ is empty because $1$ has no prime factors (it is neither prime nor composite). Thus $n=1$ is a counterexample.}
\inv{Why is $1$ excluded from being a prime number in the Fundamental Theorem of Arithmetic?}

%── B7 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction: ``There is no smallest positive rational number.''

\ans{Proof by contradiction}
\method{Assume toward a contradiction that $r \in \mathbb{Q}^+$ is the smallest positive rational number. Since $r \in \mathbb{Q}^+$, $r = a/b$ for positive integers $a,b$. Consider $q = r/2 = a/(2b)$. Since $a, 2b$ are positive integers, $q \in \mathbb{Q}^+$. Furthermore, since $r > 0$, we have $0 < q < r$. This contradicts $r$ being the smallest positive rational number. Hence, no smallest positive rational number exists.}
\inv{Does this property (density) hold for integers?}

%── B8 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``If $p$ and $q$ are distinct primes, then $p+q$ is composite.''

\ans{$p=2, q=3$}
\method{$2$ and $3$ are distinct primes, but $p+q = 2+3 = 5$, which is prime (not composite).}
\inv{If $p$ and $q$ are both odd distinct primes, is $p+q$ always composite?}

%── B9 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``For every positive integer $n\geq1$, $2^n-1$ is prime.''

\ans{$n=4$ (or $n=1, n=6$)}
\method{For $n=4$, $2^4 - 1 = 15 = 3 \times 5$, which is composite.}
\inv{Primes of the form $2^p-1$ are called Mersenne primes. Must $n$ be prime for $2^n-1$ to be prime?}

%── B10 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction: ``If $n$ is a positive integer, $n$ and $n+1$ share no common factor greater than $1$.''

\ans{Proof by contradiction}
\method{Suppose toward a contradiction that there exists a common factor $d > 1$ of $n$ and $n+1$. Then $d \mid n$ and $d \mid (n+1)$. Therefore, $d \mid ((n+1) - n) \implies d \mid 1$. But the only positive divisor of $1$ is $1$, which contradicts $d > 1$. Thus, $\gcd(n, n+1) = 1$.}
\inv{Generalise this: What is the maximum possible common factor of $n$ and $n+k$?}

\end{enumerate}

\section*{Section C}

\begin{enumerate}

%── C1 ──────────────────────────────────────────────────────────────────────────
\item Consider the statement: ``$(*)$ If a whole number $n$ is $2$ more or $4$ more than a multiple of $5$, then $n$ is prime.'' How many counterexamples to $(*)$ are there in the range $1\leq n\leq50$?

\ans{$13$}
\method{\textit{\small(after TMUA 2019 Paper 2 Q4)} Numbers in $1 \le n \le 50$ of the form $5k+2$ are: $2, 7, 12, 17, 22, 27, 32, 37, 42, 47$. Of these, the composites are $12, 22, 27, 32, 42$ ($5$ counterexamples).\\
Numbers of the form $5k+4$ are: $4, 9, 14, 19, 24, 29, 34, 39, 44, 49$. Of these, the composites are $4, 9, 14, 24, 34, 39, 44, 49$ ($8$ counterexamples).\\
Total counterexamples $= 5 + 8 = 13$.}
\inv{Why is $2$ not a counterexample despite being $5(0)+2$?}

%── C2 ──────────────────────────────────────────────────────────────────────────
\item Find the smallest positive integer $n$ for which $n^2+n+41$ is \emph{not} prime.

\ans{$40$}
\method{For $n=40$, $n^2+n+41 = 40^2+40+41 = 40(40+1)+41 = 40(41)+41 = 41(40+1) = 41^2 = 1681$, which is composite. For all $1 \le n \le 39$, $n^2+n+41$ is prime.}
\inv{Is $n=41$ also a counterexample?}

%── C3 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``$n^2+1$ is always either prime or a perfect square, for every positive integer $n$.''

\ans{$n=3$ (or $n=7, 8$)}
\method{For $n=3$, $n^2+1 = 3^2+1 = 10$. $10$ is not prime ($10 = 2 \times 5$) and $10$ is not a perfect square ($3^2 < 10 < 4^2$). Thus $n=3$ disproves the claim.}
\inv{For $n=7$, what is $n^2+1$?}

%── C4 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction: ``There is no integer solution to $x^2-3y^2=2$.''

\ans{Proof by contradiction}
\method{Suppose toward a contradiction that there exist integers $x,y$ such that $x^2 - 3y^2 = 2$. Taking this equation modulo $3$, we get $x^2 \equiv 2 \pmod 3$. However, for any integer $x$, $x \equiv 0, 1, \text{ or } 2 \pmod 3$, which gives $x^2 \equiv 0, 1, \text{ or } 1 \pmod 3$. Thus $x^2 \pmod 3 \in \{0, 1\}$, so $x^2 \equiv 2 \pmod 3$ has no solutions. Contradiction.}
\inv{Which other integer remainders cannot be quadratic residues?}

%── C5 ──────────────────────────────────────────────────────────────────────────
\item Find the smallest positive integer $n$ for which $3^n+2$ is \emph{not} prime.

\ans{$5$}
\method{Testing $n=1,2,3,4,5$:\\
$n=1 \implies 3^1+2 = 5$ (prime);\\
$n=2 \implies 3^2+2 = 11$ (prime);\\
$n=3 \implies 3^3+2 = 29$ (prime);\\
$n=4 \implies 3^4+2 = 83$ (prime);\\
$n=5 \implies 3^5+2 = 243+2 = 245 = 5 \times 49$ (composite). Smallest $n$ is $5$.}
\inv{Why is $3^n+2$ always divisible by $5$ when $n$ is an odd multiple of $5$?}

%── C6 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``If $x$ and $y$ are both non-integers, then $x+y$ is a non-integer.''

\ans{$x = 0.5, y = 0.5$}
\method{$x = 0.5 \notin \mathbb{Z}$ and $y = 0.5 \notin \mathbb{Z}$, but $x+y = 1.0 = 1 \in \mathbb{Z}$, which is an integer.}
\inv{Find a counterexample where $x$ and $y$ are not rational numbers.}

%── C7 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction: ``If $a,b$ are positive integers and $a^2+b^2$ is odd, then $a$ and $b$ do not have the same parity.''

\ans{Proof by contradiction}
\method{Suppose toward a contradiction that $a$ and $b$ have the same parity. \\
Case 1: Both $a,b$ are even. Then $a=2k, b=2m \implies a^2+b^2 = 4k^2+4m^2 = 2(2k^2+2m^2)$, which is even. \\
Case 2: Both $a,b$ are odd. Then $a=2k+1, b=2m+1 \implies a^2+b^2 = (4k^2+4k+1) + (4m^2+4m+1) = 2(2k^2+2k+2m^2+2m+1)$, which is even. \\
In both cases $a^2+b^2$ is even, contradicting $a^2+b^2$ being odd. Thus $a$ and $b$ must have opposite parity.}
\inv{What can you deduce about $a^2+b^2 \pmod 4$ when $a$ and $b$ have opposite parity?}

%── C8 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``$n^2-n+17$ is prime for every positive integer $n$.''

\ans{$n=17$ (or $n=18$)}
\method{For $n=17$, $n^2-n+17 = 17^2 - 17 + 17 = 17^2 = 289$, which is composite ($17 \times 17$).}
\inv{What is the value of $n^2-n+17$ for $n=18$?}

\end{enumerate}

\section*{Section D}

\begin{enumerate}

%── D1 ──────────────────────────────────────────────────────────────────────────
\item Consider: I. ``$n=41$ is a counterexample to `$n^2+n+41$ is always prime'.'' II. ``$n=40$ is the smallest counterexample to `$n^2+n+41$ is always prime'.'' III. ``$n=4$ is a counterexample to `$n^2+n+41$ is always prime'.'' Which of I, II, III are true?

\ans{E) I and II only}
\method{I: $41^2+41+41 = 41(41+1+1) = 41 \times 43$, composite. So I is true.\\
II: For $1 \le n \le 39$, $n^2+n+41$ is prime. At $n=40$, $40^2+40+41 = 41^2$, composite. Thus $n=40$ is indeed the smallest positive integer counterexample. II is true.\\
III: For $n=4$, $4^2+4+41 = 61$, which is prime. So $n=4$ is not a counterexample. III is false.\\
Hence, I and II only are true.}
\inv{Why does Euler's polynomial $n^2+n+41$ yield primes for all $n$ from $0$ to $39$?}

%── D2 ──────────────────────────────────────────────────────────────────────────
\item Prove by contradiction: ``There do not exist positive integers $a,b$ with $a^2-b^2=1$.''

\ans{Proof by contradiction}
\method{Assume toward a contradiction that there exist positive integers $a,b$ such that $a^2 - b^2 = 1$. Factorising as a difference of two squares gives $(a-b)(a+b) = 1$. Since $a,b \in \mathbb{Z}^+$, we have $a \ge 1$ and $b \ge 1$, so $a+b \ge 2$. However, the only factorisation of $1$ into positive integers is $1 \times 1$, requiring $a+b = 1$. This contradicts $a+b \ge 2$. Thus, no such positive integers exist.}
\inv{Do non-zero integer solutions exist if $a,b$ are not restricted to positive integers?}

%── D3 ──────────────────────────────────────────────────────────────────────────
\item A student claims: ``For every positive integer $n$, at least one of $n,n+2,n+4$ is divisible by $3$.'' Determine, with proof, whether this claim is true.

\ans{True}
\method{Proof: Any positive integer $n$ can be written as $3k$, $3k+1$, or $3k+2$ for some integer $k \ge 0$.\\
Case 1: If $n = 3k$, then $n$ is divisible by $3$.\\
Case 2: If $n = 3k+1$, then $n+2 = (3k+1)+2 = 3k+3 = 3(k+1)$, which is divisible by $3$.\\
Case 3: If $n = 3k+2$, then $n+4 = (3k+2)+4 = 3k+6 = 3(k+2)$, which is divisible by $3$.\\
In all cases, at least one of $n, n+2, n+4$ is divisible by $3$. The claim is true.}
\inv{Can more than one of $n, n+2, n+4$ be divisible by $3$ for the same $n$?}

%── D4 ──────────────────────────────────────────────────────────────────────────
\item Give a counterexample to: ``If $p$ is prime, then $2^p+1$ is prime.''

\ans{$p=3$ (or $p=5$)}
\method{For $p=3$ (which is prime), $2^p+1 = 2^3+1 = 8+1 = 9 = 3^2$, which is composite. (Also for $p=5$, $2^5+1 = 33 = 3 \times 11$).}
\inv{For which prime $p$ is $2^p+1$ actually prime?}

%── D5 ──────────────────────────────────────────────────────────────────────────
\item Statement $(*)$: ``For every positive integer $n\geq2$, there is a prime strictly between $n$ and $2n$.'' (a) Verify $(*)$ for $n=2,3,4,5$ by direct computation. (b) Does this verification constitute a proof of $(*)$? Justify your answer.

\ans{(a) $n=2: 3$; $n=3: 5$; $n=4: 5$ or $7$; $n=5: 7$. (b) No, finite checking does not prove a universal statement.}
\method{(a) For $n=2$, prime $3 \in (2,4)$. For $n=3$, prime $5 \in (3,6)$. For $n=4$, prime $5 \in (4,8)$. For $n=5$, prime $7 \in (5,10)$. \\
(b) No. Verification for finitely many cases ($n=2,3,4,5$) does not constitute a proof for infinitely many positive integers $n \ge 2$. A single unchecked value could be a counterexample. (The statement is true for all $n \ge 2$, known as Bertrand's Postulate, but proving it requires a general argument).}
\inv{What is the formal name of the theorem stating that a prime always exists between $n$ and $2n$?}

\end{enumerate}
