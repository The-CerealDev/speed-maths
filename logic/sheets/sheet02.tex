\section*{Section A \quad Rapid Recognition \hfill{\normalfont\small($\leq$15\,s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Construct a converse/contrapositive/inverse, or classify a condition. Fluency, not first contact.}

\begin{enumerate}

\item State the converse of: ``If $n$ is a multiple of $6$, then $n$ is a multiple of $3$.''

\item State the contrapositive of: ``If $n$ is a multiple of $6$, then $n$ is a multiple of $3$.''

\item True or false: ``$x=3$'' is sufficient for ``$x^2=9$''.

\item True or false: ``$x=3$'' is necessary for ``$x^2=9$''.

\item Is ``$n$ is a multiple of $4$'' necessary, sufficient, or neither, for ``$n$ is a multiple of $2$''?

\item State the inverse of: ``If $n$ is prime, then $n$ is odd or $n=2$.''

\item True or false: ``The contrapositive of a true statement is always true.''

\item True or false: ``The converse of a true statement is always true.''

\item Negate: ``If $n$ is even, then $n^2$ is even.''

\item True or false: ``$n$ is a multiple of $12$'' is sufficient for ``$n$ is a multiple of $4$''.

\end{enumerate}

\section*{Section B \quad Manipulation Drills \hfill{\normalfont\small($\leq$50\,s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Build converse/contrapositive/inverse together, and classify with justification.}

\begin{enumerate}

\item State both the converse and contrapositive of ``If $x>2$, then $x^2>4$.'' Determine the truth of each.

\item Classify ``$n=6$'' as necessary, sufficient, both, or neither, for ``$n$ is a multiple of $3$''.

\item Classify ``$n$ is divisible by $6$'' as necessary, sufficient, both, or neither, for ``$n$ is divisible by $2$ and by $3$''.

\item State the contrapositive of ``If $a$ and $b$ are both even, then $a+b$ is even.'' Is it true?

\item True or false: ``If $P\Rightarrow Q$ and $Q\Rightarrow P$ are both true, then $P$ and $Q$ are necessary and sufficient for each other.''

\item Given that ``$x$ is rational'' is sufficient for ``$x^2$ is rational'', state the converse. Is the converse true?

\item State the inverse and the converse of ``If $n$ is prime, then $n$ is odd or $n=2$.'' Are they logically equivalent to each other?

\item True or false: ``$a=b$'' is necessary and sufficient for ``$a^2=b^2$''.

\item Is ``$x^2>4$'' necessary, sufficient, both, or neither, for ``$x>2$''?

\item State the converse of ``If $n$ is a multiple of $9$, then $n$ is a multiple of $3$.'' Is the converse true?

\end{enumerate}

\section*{Section C \quad Substitution \& Structure \hfill{\normalfont\small($\leq$90\,s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Chain necessary/sufficient claims, and use the inverse--converse link.}

\begin{enumerate}

\item For real numbers $x,y$, the condition ``$x+y$ is an integer and $x-y$ is an integer'' is:
\begin{itemize}
\item[A)] necessary but not sufficient for ``$x$ and $y$ are both integers''
\item[B)] sufficient but not necessary for ``$x$ and $y$ are both integers''
\item[C)] necessary and sufficient for ``$x$ and $y$ are both integers''
\item[D)] neither necessary nor sufficient for ``$x$ and $y$ are both integers''
\end{itemize}

\item Given that $P$ is necessary and sufficient for $Q$, and $Q$ is necessary and sufficient for $R$, prove that $P$ is necessary and sufficient for $R$.

\item Classify ``a triangle has all three altitudes of equal length'' as a condition for ``the triangle is equilateral''. Justify using the fact that an altitude to side $a$ has length $\frac{2\times\text{Area}}{a}$.

\item For positive integers $a,b,n$, suppose you're told only that $n$ divides $a$, or $n$ divides $b$ (possibly both). Does this pin down whether $n$ divides the product $ab$? And working the other way: does knowing $n$ divides $ab$ pin down whether $n$ divides $a$ or $n$ divides $b$? \textit{\small(after TMUA 2019 Paper 2 Q5)}
\begin{itemize}
\item[A)] ``$n$ divides $a$ or $n$ divides $b$'' is necessary but not sufficient for ``$n$ divides $ab$''
\item[B)] ``$n$ divides $a$ or $n$ divides $b$'' is sufficient but not necessary for ``$n$ divides $ab$''
\item[C)] ``$n$ divides $a$ or $n$ divides $b$'' is necessary and sufficient for ``$n$ divides $ab$''
\item[D)] ``$n$ divides $a$ or $n$ divides $b$'' is neither necessary nor sufficient for ``$n$ divides $ab$''
\end{itemize}

\item Statement $S$: ``If $n$ is a multiple of $8$, then $n$ is a multiple of $2$.'' Determine whether $S$'s converse is true by testing $S$'s inverse instead. State the inverse, and your conclusion about the converse.

\item Classify ``$n^2$ is even'' as a condition for ``$n$ is even''.
\begin{itemize}
\item[A)] necessary but not sufficient
\item[B)] sufficient but not necessary
\item[C)] necessary and sufficient
\item[D)] neither necessary nor sufficient
\end{itemize}

\item Statement $S$ is true, and $S$'s inverse is also true. What can you conclude about $S$'s converse? Justify.

\item State the contrapositive of ``If $a$ and $b$ are both irrational, then $ab$ is irrational.'' Is the original statement true? Give a counterexample if false.

\end{enumerate}

\section*{Section D \quad Challenge \hfill{\normalfont\small(TMUA / SMC difficulty)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Insight over computation. Each question has a short, elegant solution --- find it.}

\begin{enumerate}

\item Let $f(x)=x^2+bx+c$ with $f(0)=c$. Let $P$ be the statement: ``if $c<0$, then $f$ has two distinct real roots.'' Which of the following must be true? \textit{\small(after TMUA 2022 Paper 2 Q5)}
\begin{itemize}
\item[I.] $P$
\item[II.] The converse of $P$
\item[III.] The contrapositive of $P$
\end{itemize}
\begin{itemize}
\item[A)] none of them
\item[B)] I only
\item[C)] II only
\item[D)] III only
\item[E)] I and II only
\item[F)] I and III only
\item[G)] II and III only
\item[H)] I, II and III
\end{itemize}

\item Consider a finite list of $n$ real numbers (repeats allowed), sorted into non-decreasing order. Let $P$ be the statement ``$n$ is odd'', and let $Q$ be the statement ``the middle value of the sorted list equals one of the list's own entries''. $P$ is: \textit{\small(after TMUA 2022 Paper 2 Q6)}
\begin{itemize}
\item[A)] necessary but not sufficient for $Q$
\item[B)] sufficient but not necessary for $Q$
\item[C)] necessary and sufficient for $Q$
\item[D)] neither necessary nor sufficient for $Q$
\end{itemize}

\item For a positive integer $n$, let $P$ be ``$n$ leaves remainder $1$ when divided by $4$'' and $Q$ be ``$n^2$ leaves remainder $1$ when divided by $8$.'' $P$ is:
\begin{itemize}
\item[A)] necessary but not sufficient for $Q$
\item[B)] sufficient but not necessary for $Q$
\item[C)] necessary and sufficient for $Q$
\item[D)] neither necessary nor sufficient for $Q$
\end{itemize}

\item Statement $S$: ``If a quadrilateral has four equal sides, then its diagonals are perpendicular.'' (a) Is $S$ true? (b) State $S$'s converse. Is the converse true?

\item Statement $S$: ``If $n$ is a positive integer such that $n^2-1$ is prime, then $n=2$.'' (a) Is $S$ true? Prove or disprove. (b) State $S$'s contrapositive. (c) Is ``$n^2-1$ is prime'' necessary and sufficient for ``$n=2$''?

\end{enumerate}
