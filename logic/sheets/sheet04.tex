\SpeedTitleBlock{Daily Logic Drill \#4}{David Akinyele-Aje}
\noindent\textit{New toolkit today: Disproof by Counterexample $\cdot$ Proof by Contradiction.}

\vspace{2pt}
\noindent\textit{Attempt each question before checking answers. Time yourself. No calculators.}

\section*{Section A \quad Rapid Recognition \hfill{\normalfont\small($\leq$15\,s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Kill a universal claim with a single well-chosen example. Fluency, not first contact.}

\begin{enumerate}

\item Give a counterexample to: ``Every prime number is odd.''

\item Give a counterexample to: ``$n^2\geq n$ for every real number $n$.''

\item Give a counterexample to: ``If $n$ is prime, then $n+2$ is prime.''

\item Give a counterexample to: ``Every multiple of $4$ is a multiple of $8$.''

\item To prove ``there is no largest even integer'' by contradiction, what should you assume first?

\item True or false: ``If assuming `not $P$' leads to something impossible, $P$ must be true.''

\item Give a counterexample to: ``$\sqrt{a+b}=\sqrt a+\sqrt b$ for all $a,b\geq0$.''

\item Give a counterexample to: ``$(x+y)^2=x^2+y^2$ for all real $x,y$.''

\item True or false: ``A single counterexample is enough to disprove a `for all' claim.''

\item Give a counterexample to: ``$n!+1$ is always prime.''

\end{enumerate}

\section*{Section B \quad Manipulation Drills \hfill{\normalfont\small($\leq$50\,s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Hunt systematically, and write your first two contradiction proofs.}

\begin{enumerate}

\item Five envelopes contain the numbers $6,9,14,21,25$ respectively. Each is sealed with a wax stamp showing either a star or a circle: the envelopes containing $6,14,25$ are star-sealed, and the envelopes containing $9,21$ are circle-sealed. Claim: ``Every star-sealed envelope contains an even number.'' Which envelope's contents provide a counterexample?

\item Give a counterexample to: ``If $a$ and $b$ are both irrational, then $a+b$ is irrational.''

\item Give a counterexample to: ``If $a$ and $b$ are both irrational, then $ab$ is irrational.''

\item Prove by contradiction: ``There is no largest even integer.''

\item Give a counterexample (using $n=0$) to: ``For every integer $n$, $n/n=1$.''

\item Give a counterexample (using $n=1$) to: ``Every positive integer has at least one prime factor.''

\item Prove by contradiction: ``There is no smallest positive rational number.''

\item Give a counterexample to: ``If $p$ and $q$ are distinct primes, then $p+q$ is composite.''

\item Give a counterexample to: ``For every positive integer $n\geq1$, $2^n-1$ is prime.''

\item Prove by contradiction: ``If $n$ is a positive integer, $n$ and $n+1$ share no common factor greater than $1$.''

\end{enumerate}

\section*{Section C \quad Substitution \& Structure \hfill{\normalfont\small($\leq$90\,s each)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Find the smallest counterexample, not just any counterexample --- and write longer contradiction proofs.}

\begin{enumerate}

\item Consider the statement: ``$(*)$ If a whole number $n$ is $2$ more or $4$ more than a multiple of $5$, then $n$ is prime.'' How many counterexamples to $(*)$ are there in the range $1\leq n\leq50$? \textit{\small(after TMUA 2019 Paper 2 Q4)}

\item Find the smallest positive integer $n$ for which $n^2+n+41$ is \emph{not} prime.

\item Give a counterexample to: ``$n^2+1$ is always either prime or a perfect square, for every positive integer $n$.''

\item Prove by contradiction: ``There is no integer solution to $x^2-3y^2=2$.''

\item Find the smallest positive integer $n$ for which $3^n+2$ is \emph{not} prime.

\item Give a counterexample to: ``If $x$ and $y$ are both non-integers, then $x+y$ is a non-integer.''

\item Prove by contradiction: ``If $a,b$ are positive integers and $a^2+b^2$ is odd, then $a$ and $b$ do not have the same parity.''

\item Give a counterexample to: ``$n^2-n+17$ is prime for every positive integer $n$.''

\end{enumerate}

% ══════════════════════════════════════════════════════════════════════════════
\section*{Section D \quad Challenge \hfill{\normalfont\small(TMUA / SMC difficulty)}}
% ══════════════════════════════════════════════════════════════════════════════

\noindent\textit{Insight over computation. Each question has a short, elegant solution --- find it.}

\begin{enumerate}

\item Consider: I. ``$n=41$ is a counterexample to `$n^2+n+41$ is always prime'.'' II. ``$n=40$ is the smallest counterexample to `$n^2+n+41$ is always prime'.'' III. ``$n=4$ is a counterexample to `$n^2+n+41$ is always prime'.'' Which of I, II, III are true?
\begin{itemize}
\item[A)] none
\item[B)] I only
\item[C)] II only
\item[D)] III only
\item[E)] I and II only
\item[F)] I and III only
\item[G)] II and III only
\item[H)] I, II and III
\end{itemize}

\item Prove by contradiction: ``There do not exist positive integers $a,b$ with $a^2-b^2=1$.''

\item A student claims: ``For every positive integer $n$, at least one of $n,n+2,n+4$ is divisible by $3$.'' Determine, with proof, whether this claim is true.

\item Give a counterexample to: ``If $p$ is prime, then $2^p+1$ is prime.''

\item Statement $(*)$: ``For every positive integer $n\geq2$, there is a prime strictly between $n$ and $2n$.'' (a) Verify $(*)$ for $n=2,3,4,5$ by direct computation. (b) Does this verification constitute a proof of $(*)$? Justify your answer.

\end{enumerate}
