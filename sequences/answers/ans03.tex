\section*{Section A \quad Rapid Recognition}
%═══════════════════════════════════════════════════════════════════════════════
\begin{enumerate}

%── A1 ──────────────────────────────────────────────────────────────────────────
\item Write down the common ratio of the GP $3,6,12,24,\ldots$

\ans{$2$}
\method{$6/3=2$, $12/6=2$, $24/12=2$ --- constant ratio $2$.}
\inv{A GP's ratio is \emph{next} $\div$ \emph{current}, never the reverse. \textbf{So what single equation must three numbers $a,b,c$ satisfy to sit consecutively in a GP?} Derive it from the two ratios being equal, check it on $3,6,12$, and note that it makes no reference to the ratio itself.}

%── A2 ──────────────────────────────────────────────────────────────────────────
\item A GP has first term $5$ and common ratio $2$. Find the 6th term.

\ans{$160$}
\method{$a_6=5\times2^5=5\times32=160$.}
\inv{Compare $a+(n-1)d$ with $ar^{n-1}$: multiplication replaces addition throughout. \textbf{For $a>0$, exactly which $r$ make the GP strictly increasing? Now answer the same question for $a<0$.} The two answers are different, and the sign of $a$ is the whole reason.}

%── A3 ──────────────────────────────────────────────────────────────────────────
\item A GP has first term $2$ and common ratio $3$. Find $S_5$.

\ans{$242$}
\method{$S_5=\dfrac{2(3^5-1)}{3-1}=\dfrac{2\times242}{2}=242$.}
\inv{Check directly: $2+6+18+54+162=242$ \checkmark. \textbf{Notice $242=3^5-1$. Prove that for this GP $S_n=3^n-1$ exactly, for every $n$.} Substitute $a=2,r=3$ into the sum formula and watch the $2$ cancel --- then say what it is about $a$ and $r$ here that makes the answer this clean.}

%── A4 ──────────────────────────────────────────────────────────────────────────
\item State the condition on $r$ for an infinite GP's sum to converge.

\ans{$|r|<1$}
\method{The infinite sum $a+ar+ar^2+\cdots$ only settles to a finite value when each term shrinks toward $0$, which needs $|r|<1$.}
\inv{The one condition the sheet is built around. \textbf{Test the boundary yourself: with $a=1$ and $r=-1$, write out the first six partial sums.} Do they approach a limit, oscillate, or grow? Then explain why $|r|<1$ has to be a \emph{strict} inequality on the magnitude, not on $r$ itself.}

%── A5 ──────────────────────────────────────────────────────────────────────────
\item A GP has first term $8$ and common ratio $\tfrac12$. Find its sum to infinity.

\ans{$16$}
\method{$S_\infty=\dfrac{8}{1-\frac12}=\dfrac8{\frac12}=16$.}
\inv{Partial sums $8,12,14,15,15.5,\ldots$ climb toward $16$ without reaching it. The shortfall after $n$ terms is exactly $16-S_n$. \textbf{Find a formula for that shortfall, then determine how many terms are needed to get within $0.01$ of $16$.} No logarithms required --- the shortfall is a power of $2$.}

%── A6 ──────────────────────────────────────────────────────────────────────────
\item Which of these three sequences is geometric: (i) $2,4,6,8$ \quad (ii) $2,4,8,16$ \quad (iii) $2,4,7,11$?

\ans{Sequence (ii)}
\method{(i) has constant difference $2$ --- arithmetic, not geometric. (iii) has differences $2,3,4$ and ratios $2,1.75,1.57\ldots$ --- neither.}
\inv{Test against \emph{both} definitions before concluding ``neither'' --- (iii) genuinely fits no pattern. \textbf{Can a single sequence be both arithmetic and geometric at once?} Write down what each definition demands, solve the two conditions together, and describe every sequence that satisfies both.}

%── A7 ──────────────────────────────────────────────────────────────────────────
\item Write down the sum-to-infinity formula for a GP with first term $a$ and ratio $r$ (where $|r|<1$).

\ans{$\dfrac{a}{1-r}$}
\method{Take the finite-sum formula $S_n=\dfrac{a(1-r^n)}{1-r}$ and let $n\to\infty$: since $|r|<1$, $r^n\to0$, leaving $S_\infty=\dfrac a{1-r}$.}
\inv{Taking a limit of the finite formula is exactly how D1's proof works. \textbf{Derive $S_\infty=\frac a{1-r}$ a second way, without any limit: write out $S$, multiply it by $r$, and subtract.} Then identify the exact step in that argument that silently assumes $|r|<1$ --- it is easy to miss.}

%── A8 ──────────────────────────────────────────────────────────────────────────
\item True or false: a GP with common ratio $r=1$ has a well-defined sum to infinity.

\ans{False}
\method{At $r=1$ every term equals $a$, so the sum after $n$ terms is $na$, which grows without bound (unless $a=0$) --- it never settles to a finite value.}
\inv{$r=1$ is precisely the excluded boundary. \textbf{Put $r=1$ into $\frac{a(1-r^n)}{1-r}$ and see what breaks; then work out what $S_n$ actually is for a GP with $r=1$.} The true answer is obvious from the sequence itself, which makes it a good check on whether a formula is being applied blindly.}

%── A9 ──────────────────────────────────────────────────────────────────────────
\item A GP has first term $100$ and common ratio $-\tfrac12$. Find its sum to infinity.

\ans{$\dfrac{200}3$}
\method{$S_\infty=\dfrac{100}{1-(-\frac12)}=\dfrac{100}{\frac32}=\dfrac{200}3$.}
\inv{A negative ratio \emph{increases} the denominator: $1-(-\tfrac12)=\tfrac32$. \textbf{Compare this GP's sum with the same first term but $r=+\tfrac12$. What is the ratio of the two sums, and does that factor depend on $a$ at all?} Then generalise: how do $S_\infty$ for $r$ and for $-r$ compare in general?}

%── A10 ─────────────────────────────────────────────────────────────────────────
\item Write the next term of the GP $81, 27, 9, 3, \ldots$

\ans{$1$}
\method{Ratio is $\tfrac13$: $81,27,9,3,1,\ldots$}
\inv{The terms shrink toward $0$ without reaching it --- what makes a sum to infinity possible. \textbf{But is ``terms tend to $0$'' \emph{enough} to guarantee a finite sum? Test it on $1+\tfrac12+\tfrac13+\tfrac14+\cdots$, whose terms certainly tend to $0$.} Group the terms in blocks of $2,4,8,\ldots$ and bound each block from below.}

\end{enumerate}

\section*{Section B \quad Manipulation Drills}

\begin{enumerate}

%── B1 ──────────────────────────────────────────────────────────────────────────
\item A GP has first term $4$ and common ratio $3$. Find the smallest $n$ for which $S_n>1000$.

\ans{$n=6$}
\method{$S_n=\dfrac{4(3^n-1)}{2}=2(3^n-1)>1000\Rightarrow3^n>501$. $3^5=243$ (too small), $3^6=729>501$. So $n=6$.}
\inv{Test boundary powers directly rather than reaching for logarithms under time pressure. \textbf{Here $S_n=2(3^n-1)$, so the condition reduces to a single inequality in $3^n$ --- write it down and confirm which power of $3$ first clears it.} Then state the general shortcut: for $S_n>T$, what inequality in $r^n$ are you always really solving?}

%── B2 ──────────────────────────────────────────────────────────────────────────
\item For which values of $r$ does the infinite GP with first term $5$ have a finite sum?
\begin{itemize}
\item[A)] all real $r$
\item[B)] $r>0$ only
\item[C)] $-1<r<1$
\item[D)] $r\neq1$
\end{itemize}

\ans{C) $-1<r<1$}
\method{The sum-to-infinity formula $S_\infty=\dfrac5{1-r}$ is only valid, i.e.\ only converges, when $|r|<1$ --- the first term's value ($5$) plays no role in whether convergence happens.}
\inv{Convergence depends \emph{only} on $r$, never on $a$. \textbf{So does $a$ influence anything about the infinite sum at all --- and what happens in the single special case $a=0$?} Say precisely which features of the series $a$ controls and which it does not.}

%── B3 ──────────────────────────────────────────────────────────────────────────
\item A GP has first term $3$ and common ratio $\tfrac23$. Find its sum to infinity.

\ans{$9$}
\method{$S_\infty=\dfrac3{1-\frac23}=\dfrac3{\frac13}=9$.}
\inv{The terms $3,2,\tfrac43,\tfrac89,\ldots$ have running total climbing toward $9$. \textbf{How many terms are needed before the partial sum passes half the total, i.e.\ exceeds $4.5$?} Set up the inequality in $\left(\tfrac23\right)^n$ --- the answer is far smaller than most people guess, which is itself worth noticing.}

%── B4 ──────────────────────────────────────────────────────────────────────────
\item A GP has sum to infinity $20$ and first term $5$. Find $r$.
\begin{itemize}
\item[A)] $\tfrac34$
\item[B)] $\tfrac14$
\item[C)] $\tfrac45$
\item[D)] $-\tfrac34$
\end{itemize}

\ans{A) $\tfrac34$}
\method{$20=\dfrac5{1-r}\Rightarrow1-r=\dfrac5{20}=\dfrac14\Rightarrow r=\dfrac34$.}
\inv{Isolate $(1-r)$ by cross-multiplying, then solve for $r$ last --- wrong order invites sign slips. \textbf{Now run it backwards: for a GP with $a=5$, which sums to infinity are actually achievable as $r$ ranges over $|r|<1$?} You should find the reachable values are not all of $\mathbb R$ --- describe exactly which are excluded, and why $S_\infty$ can never be small.}

%── B5 ──────────────────────────────────────────────────────────────────────────
\item A GP has first term $1$ and sum to infinity $2$. Find the second term.
\begin{itemize}
\item[A)] $\tfrac12$
\item[B)] $\tfrac14$
\item[C)] $2$
\item[D)] $1$
\end{itemize}

\ans{A) $\tfrac12$}
\method{$2=\dfrac1{1-r}\Rightarrow1-r=\dfrac12\Rightarrow r=\dfrac12$. Second term $=1\times\dfrac12=\dfrac12$.}
\inv{The question asks for the second \emph{term}, not the ratio --- here they coincide numerically, which is exactly the trap. \textbf{Change the first term to $a=3$ with the same sum to infinity $2$: does the coincidence survive?} Then say for which $a$ the second term equals the common ratio.}

%── B6 ──────────────────────────────────────────────────────────────────────────
\item An AP and a GP both have first term $4$. The AP has common difference $2$; the GP has common ratio $2$. Which sequence has the larger 10th term?
\begin{itemize}
\item[A)] The AP
\item[B)] The GP
\item[C)] They are equal
\item[D)] Cannot be determined
\end{itemize}

\ans{B) The GP}
\method{AP 10th term: $4+9(2)=22$. GP 10th term: $4\times2^9=4\times512=2048$. $2048\gg22$.}
\inv{Geometric eventually beats arithmetic --- by the 10th term the gap is already near two orders of magnitude. \textbf{But they start equal at $n=1$. From which term onward is the GP permanently ahead, and can you prove it never falls behind again?} Induction on the ratio between consecutive terms is the clean route.}

%── B7 ──────────────────────────────────────────────────────────────────────────
\item A GP has all terms positive and common ratio $r>1$. Does its sum to infinity exist?
\begin{itemize}
\item[A)] Yes, always.
\item[B)] No, never --- it diverges.
\item[C)] Only if the first term is small.
\item[D)] Only if $r<2$.
\end{itemize}

\ans{B) No, never --- it diverges.}
\method{$r>1$ means $|r|>1$, which is exactly the case excluded from convergence --- each term is strictly bigger than the last, so the partial sums grow without bound.}
\inv{``All terms positive'' is a red herring: only $|r|$ decides convergence. \textbf{Build the sharpest counterexamples you can --- a GP with all terms positive that diverges, and one with terms of alternating sign that converges.} Then state what, if anything, the sign of the terms does tell you about the sum.}

%── B8 ──────────────────────────────────────────────────────────────────────────
\item Express the recurring decimal $0.\overline3=0.333\ldots$ as a fraction, using the geometric series sum-to-infinity formula.

\ans{$\dfrac13$}
\method{$0.\overline3=\dfrac3{10}+\dfrac3{100}+\dfrac3{1000}+\cdots$, a GP with $a=\dfrac3{10}$, $r=\dfrac1{10}$. $S_\infty=\dfrac{3/10}{1-1/10}=\dfrac{3/10}{9/10}=\dfrac39=\dfrac13$.}
\inv{Every recurring decimal is a GP with $a=$ (repeating block over its place value) and $r=10^{-(\text{block length})}$. \textbf{Use it on $0.\overline{142857}$ and see which famous fraction appears. Then answer the structural question: why must every recurring decimal be rational?} The block length controls the denominator.}

%── B9 ──────────────────────────────────────────────────────────────────────────
\item A GP has first term $2$ and common ratio $-3$. Find $S_4$.
\begin{itemize}
\item[A)] $40$
\item[B)] $-40$
\item[C)] $160$
\item[D)] $-160$
\end{itemize}

\ans{B) $-40$}
\method{$S_4=\dfrac{2((-3)^4-1)}{-3-1}=\dfrac{2(81-1)}{-4}=\dfrac{160}{-4}=-40$.}
\inv{The trap is mishandling the signed denominator $1-r=4$. \textbf{Compute $S_1,\ldots,S_5$ for this GP and describe the pattern the partial sums follow.} With $r=-3$ they neither increase nor converge --- say exactly what they do, and connect that to where $|r|=3$ sits relative to $1$.}

%── B10 ─────────────────────────────────────────────────────────────────────────
\item Recall from Day 2 that $(1-x)^{-1}=1+x+x^2+\cdots$ for $|x|<1$. Using $x=\tfrac13$, what does this series sum to?
\begin{itemize}
\item[A)] $\tfrac32$
\item[B)] $\tfrac23$
\item[C)] $3$
\item[D)] $2$
\end{itemize}

\ans{A) $\tfrac32$}
\method{$(1-x)^{-1}=\dfrac1{1-x}$. At $x=\dfrac13$: $\dfrac1{1-1/3}=\dfrac1{2/3}=\dfrac32$.}
\inv{$(1-x)^{-1}$ evaluated at a specific $x$ \emph{is} a sum-to-infinity calculation --- this is B3's GP in Day 2's notation. \textbf{So what is the exact range of $x$ for which that series identity is valid, and what is it about $x=1$ that makes both sides fail simultaneously?} Compare the two sides just inside and just outside the range.}

\end{enumerate}

\begin{enumerate}

%── C1 ──────────────────────────────────────────────────────────────────────────
\item A GP has first term $1$ and unknown common ratio $r$. Its sum to infinity exists and equals $4$. What must be true about $r$?
\begin{itemize}
\item[A)] $r=\tfrac34$ exactly.
\item[B)] $-1<r<1$, but no exact value is determinable.
\item[C)] $r=\tfrac43$.
\item[D)] Infinitely many values of $r$ satisfy this.
\end{itemize}

\ans{A) $r=\tfrac34$ exactly}
\method{$\dfrac1{1-r}=4\Rightarrow1-r=\dfrac14\Rightarrow r=\dfrac34$ --- a single equation in one unknown has exactly one solution.}
\inv{With $a$ fixed, $S_\infty=4$ pins $r$ down uniquely --- nothing is underdetermined. \textbf{Now let the target sum vary: for $a=1$, which sums to infinity are achievable, and which are impossible no matter how $r$ is chosen?} Solve $S=\frac1{1-r}$ for $r$ and impose $|r|<1$ --- the impossible band is larger than it first appears.}

%── C2 ──────────────────────────────────────────────────────────────────────────
\item A geometric sequence has first term $a$ and common ratio $r$, both positive integers with $r>1$. Let $S_n$ denote the sum of the first $n$ terms. Given $S_{18}-S_{12}=kS_6$ for some positive integer $k$, find the smallest possible value of $k$. \textit{\small(after TMUA 2022 Paper 1 Q8)}
\begin{itemize}
\item[A)] $2^6$
\item[B)] $2^{12}$
\item[C)] $2^{18}$
\item[D)] $\dfrac{2^6}{2^6-1}$
\item[E)] $2^6(2^6-1)$
\end{itemize}

\ans{B) $2^{12}$}
\method{$S_{18}-S_{12}=\dfrac{a(r^{18}-r^{12})}{r-1}=\dfrac{ar^{12}(r^6-1)}{r-1}=r^{12}\cdot\dfrac{a(r^6-1)}{r-1}=r^{12}S_6$. So $k=r^{12}$, minimised over integers $r>1$ at $r=2$: $k=2^{12}=4096$.}
\inv{The exponent is always the index gap between the two subtracted sums. \textbf{Prove the general rule: show $S_{m+j}-S_m=r^m S_j$ for any GP, then use it to predict $k$ before computing anything.} \textbf{Check your rule against the real TMUA question (2022 Paper 1 Q8)}, which uses $S_{30}-S_{20}=kS_{10}$ and minimises $k$ to $2^{20}$.}

%── C3 ──────────────────────────────────────────────────────────────────────────
\item An infinite geometric series has first term $a=6$ and common ratio $r=\dfrac{3k}{10}$, where $k$ is chosen uniformly at random from the integers $-5,-4,\ldots,4,5$ (11 equally likely values). Find the probability that the series converges \emph{and} its sum to infinity exceeds $5$. \textit{\small(after TMUA 2023 Paper 1 Q18)}
\begin{itemize}
\item[A)] $\tfrac4{11}$
\item[B)] $\tfrac5{11}$
\item[C)] $\tfrac7{11}$
\item[D)] $\tfrac3{11}$
\end{itemize}

\ans{A) $\tfrac4{11}$}
\method{Convergence needs $|r|<1$: $\left|\frac{3k}{10}\right|<1\Rightarrow|k|<\frac{10}3\approx3.33\Rightarrow k\in\{-3,\ldots,3\}$ ($7$ of the $11$ values). Among all $11$ possible $k$, $S_\infty=\dfrac6{1-3k/10}=\dfrac{60}{10-3k}>5$ requires $60>5(10-3k)\Rightarrow60>50-15k\Rightarrow15k>-10\Rightarrow k>-\tfrac23\Rightarrow k\geq0$ (since $10-3k>0$ throughout the converging range, the inequality direction is preserved). Combined with convergence, valid $k\in\{0,1,2,3\}$ --- $4$ values. Probability $=\dfrac4{11}$.}
\inv{The denominator is the \emph{full} sample space ($11$), not just the converging subset --- a non-converging $k$ fails the condition rather than leaving the count. \textbf{Confirm that by listing which $k$ converge and which of those clear the threshold. Then ask the trap question: how would the answer change if the problem said ``given that the series converges''?} \textbf{Finally try the real TMUA version (2023 Paper 1 Q18)}: $a=4$, $r=\frac{2k}7$, threshold $S>3$, answer $\frac5{11}$.}

%── C4 ──────────────────────────────────────────────────────────────────────────
\item The series $\sum_{k=1}^\infty ar^{k-1}$ converges (for real $a,r$) when $|r|<1$. If $a$ and $r$ are instead allowed to be complex numbers, which is the correct convergence condition?
\begin{itemize}
\item[A)] $|r|<1$ (the same condition, using the complex magnitude).
\item[B)] $r<1$ (a real inequality).
\item[C)] $a<1$.
\item[D)] There is no meaningful convergence condition for complex $r$.
\end{itemize}

\ans{A) $|r|<1$ (using the complex magnitude)}
\method{The geometric series' convergence proof relies only on $|r^n|\to0$, and $|r^n|=|r|^n$ holds identically whether $r$ is real or complex --- so the same magnitude condition transfers directly.}
\inv{Convergence is about the \emph{size} of $r$ shrinking powers to $0$ --- a notion indifferent to whether $r$ is real or complex. \textbf{So describe the set of complex $r$ for which the series converges, as a region of the plane. Then test $r=i$: what do the partial sums do, and is $|i|<1$?}}

%── C5 ──────────────────────────────────────────────────────────────────────────
\item Convert the recurring base-2 number $0.\overline{0111}=0.0111\,0111\,0111\ldots$ (base $2$) to a fraction in base $10$. \textit{\small(after TMUA 2023 Paper 2 Q15)}
\begin{itemize}
\item[A)] $\tfrac13$
\item[B)] $\tfrac15$
\item[C)] $\tfrac7{15}$
\item[D)] $\tfrac7{16}$
\end{itemize}

\ans{C) $\dfrac7{15}$}
\method{The repeating block $0111$ (base $2$) has integer value $0\times8+1\times4+1\times2+1\times1=7$, over a $4$-bit block. This is a GP with $a=\dfrac7{16}$, $r=\dfrac1{16}$: $S_\infty=\dfrac{7/16}{1-1/16}=\dfrac{7/16}{15/16}=\dfrac7{15}$.}
\inv{Same mechanism as B8, with block value and place value both read in base $2$. \textbf{State the general base-$b$ rule: a block of length $L$ with value $v$ recurring gives which fraction?} Then check it reproduces both B8 (base $10$) and this question, \textbf{and try the real TMUA version (2023 Paper 2 Q15), block $0011$, answer $\tfrac15$}.}

%── C6 ──────────────────────────────────────────────────────────────────────────
\item Which of the following statements about arithmetic and geometric series is FALSE?
\begin{itemize}
\item[A)] An AP's terms grow linearly in $n$; a GP's terms with $|r|>1$ grow exponentially in $n$.
\item[B)] An AP diverges to $\pm\infty$ (or stays constant) as $n\to\infty$, unless $d=0$.
\item[C)] A GP can converge to a finite sum to infinity even though no individual term is exactly zero, provided $|r|<1$.
\item[D)] Every convergent GP has all its terms equal to zero eventually.
\end{itemize}

\ans{D) Every convergent GP has all its terms equal to zero eventually.}
\method{A convergent GP's terms $ar^{n-1}\to0$ as a \emph{limit}, but for any finite $n$ and $r\neq0$, $ar^{n-1}\neq0$ exactly --- the terms get arbitrarily small without ever reaching zero. A), B), C) are all genuinely true statements.}
\inv{The same ``limit vs.\ attained'' idea as $0.\overline9=1$: neither the terms' $0$ nor the sum's $S_\infty$ is reached at any finite step. \textbf{So identify the one degenerate case in which a GP's partial sums \emph{do} literally attain $S_\infty$ after finitely many terms.} There is exactly one such $r$, and it makes the series stop rather than converge.}

%── C7 ──────────────────────────────────────────────────────────────────────────
\item Using $(1-x)^{-1}=\sum_{k=0}^\infty x^k$ from Day 2, evaluate $\displaystyle\sum_{k=0}^\infty\dfrac1{5^k}$ by choosing an appropriate value of $x$.
\begin{itemize}
\item[A)] $\tfrac54$
\item[B)] $\tfrac45$
\item[C)] $5$
\item[D)] $\tfrac14$
\end{itemize}

\ans{A) $\tfrac54$}
\method{$\displaystyle\sum_{k=0}^\infty\frac1{5^k}=\sum_{k=0}^\infty\left(\frac15\right)^k=(1-\tfrac15)^{-1}$ at $x=\tfrac15$: $(1-\tfrac15)^{-1}=(\tfrac45)^{-1}=\dfrac54$.}
\inv{Recognising $\sum\frac1{5^k}$ as $\sum x^k$ at $x=\tfrac15$ makes it a one-line substitution. \textbf{Now generalise: $\sum_{k=0}^\infty\frac1{b^k}$ for an integer $b\geq2$ has a strikingly simple closed form --- find it, and say what it tends to as $b$ grows.} Test it at $b=2$ against a sum you already know.}

%── C8 ──────────────────────────────────────────────────────────────────────────
\item Let $S=1+2x+3x^2+4x^3+\cdots$. Recall from Day 2 that this is exactly $(1-x)^{-2}=\sum_{k=0}^\infty(k+1)x^k$. Evaluate $S$ at $x=\tfrac14$.
\begin{itemize}
\item[A)] $\tfrac{16}9$
\item[B)] $\tfrac43$
\item[C)] $\tfrac9{16}$
\item[D)] $\tfrac34$
\end{itemize}

\ans{A) $\tfrac{16}9$}
\method{$(1-x)^{-2}$ at $x=\tfrac14$: $(1-\tfrac14)^{-2}=(\tfrac34)^{-2}=\left(\tfrac43\right)^2=\dfrac{16}9$.}
\inv{$1+2x+3x^2+\cdots$ is the term-by-term derivative of $1+x+x^2+\cdots$. \textbf{Push it one step further: differentiate again to find a closed form for $\sum_{k}(k+1)(k+2)x^k$, and check your result at $x=\tfrac14$ against this question's answer.} Then predict the closed form of $(1-x)^{-m}$ without differentiating $m$ times.}

\end{enumerate}

\section*{Section D \quad Challenge}
%═══════════════════════════════════════════════════════════════════════════════
\begin{enumerate}

%── D1 ──────────────────────────────────────────────────────────────────────────
\item Prove that $\displaystyle\sum_{k=1}^\infty\dfrac1{2^k}=1$ exactly (not merely ``approaches $1$''), using the sum-to-infinity formula. Then explain why the partial sum $S_n=\sum_{k=1}^n\dfrac1{2^k}=1-\dfrac1{2^n}$ never actually equals $1$ for any finite $n$.

\ans{Proof via the sum-to-infinity formula; partial sums approach but never reach $1$.}
\method{$a=\frac12,r=\frac12$: $S_\infty=\dfrac{1/2}{1-1/2}=\dfrac{1/2}{1/2}=1$ exactly, by the formula itself (not an approximation). Separately, $S_n=\dfrac{a(1-r^n)}{1-r}=\dfrac{\frac12(1-(\frac12)^n)}{\frac12}=1-\dfrac1{2^n}$, which is strictly less than $1$ for every finite $n$ (since $\frac1{2^n}>0$ always) --- the infinite sum is the \emph{limit} of these partial sums, not a partial sum itself.}
\inv{The infinite sum is \emph{defined} as the limit, and that limit is exactly $1$ --- no finite partial sum ever is. \textbf{Make the definition do visible work: given any tolerance $\varepsilon>0$, find explicitly how many terms guarantee $1-S_n<\varepsilon$.} Producing that $n$ from $\varepsilon$ is precisely what ``the limit is $1$'' asserts.}

%── D2 ──────────────────────────────────────────────────────────────────────────
\item Let $x=0.\overline9=0.999\ldots$ (a recurring $9$). Writing $x$ as an infinite geometric series and summing it, determine $x$ exactly.
\begin{itemize}
\item[A)] $x=1$ exactly.
\item[B)] $x$ is infinitesimally less than $1$ but not equal to it.
\item[C)] $0.\overline9$ is undefined.
\item[D)] $x=\tfrac9{10}$.
\end{itemize}

\ans{A) $x=1$ exactly.}
\method{$0.\overline9=\dfrac9{10}+\dfrac9{100}+\cdots$, a GP with $a=\dfrac9{10},r=\dfrac1{10}$. $S_\infty=\dfrac{9/10}{1-1/10}=\dfrac{9/10}{9/10}=1$.}
\inv{There is no sense in which $0.\overline9$ is ``slightly less than $1$'': it is the limit of its partial sums, and that limit is $1$. \textbf{Test the objection directly --- if $0.\overline9\neq1$, their difference is a positive number, so name it.} Show no positive real can be smaller than every $10^{-n}$, and the objection collapses.}

%── D3 ──────────────────────────────────────────────────────────────────────────
\item A ball is dropped from height $h$. Each time it bounces, it rebounds to a fraction $r$ of its previous height ($0<r<1$). Find, in terms of $h$ and $r$, the total distance the ball travels (up and down, forever) before coming to rest.
\begin{itemize}
\item[A)] $h\cdot\dfrac{1+r}{1-r}$
\item[B)] $h\cdot\dfrac2{1-r}$
\item[C)] $h\cdot\dfrac1{1-r}$
\item[D)] $h(1+2r)$
\end{itemize}

\ans{A) $h\cdot\dfrac{1+r}{1-r}$}
\method{Total distance $=$ (initial drop $h$) $+$ (every rebound-and-fall pair, each pair covering $2\times$ that rebound's height). Rebound heights are $hr,hr^2,hr^3,\ldots$, a GP with first term $hr$, ratio $r$. Distance from rebounds $=2\times\dfrac{hr}{1-r}$. Total $=h+\dfrac{2hr}{1-r}=\dfrac{h(1-r)+2hr}{1-r}=\dfrac{h(1+r)}{1-r}$.}
\inv{The factor of $2$ per rebound is the easiest thing to forget --- separate the unpaired initial drop from the infinitely many paired bounces. \textbf{Now sanity-check the formula at its two extremes: what total distance does it predict as $r\to0$, and as $r\to1^-$?} Both limits should match physical intuition; if one doesn't, the factor of $2$ is in the wrong place.}

%── D4 ──────────────────────────────────────────────────────────────────────────
\item The sum to infinity of a GP equals the sum of its first two terms. What can be said about the common ratio $r$?
\begin{itemize}
\item[A)] $r=0$ (the ``GP'' is actually constant).
\item[B)] $r=\pm1$.
\item[C)] $r=\tfrac12$.
\item[D)] No such $r$ exists.
\end{itemize}

\ans{A) $r=0$}
\method{$S_\infty=a(1+r)\Rightarrow\dfrac a{1-r}=a(1+r)\Rightarrow\dfrac1{1-r}=1+r$ (for $a\neq0$) $\Rightarrow1=(1+r)(1-r)=1-r^2\Rightarrow r^2=0\Rightarrow r=0$.}
\inv{$r=0$ collapses the GP to its first term --- degenerate, but the algebra forces it uniquely. \textbf{Try the natural follow-up: for which $r$ does the sum to infinity equal the sum of the first \emph{three} terms?} Set up the equation as before and see whether it again forces $r=0$, or whether new solutions appear.}

%── D5 ──────────────────────────────────────────────────────────────────────────
\item A sequence is defined by $a_1=3$ and $a_{n+1}=\tfrac12a_n$ for $n\geq1$. Using the sum-to-infinity formula, evaluate $\displaystyle\sum_{n=1}^\infty a_n$.
\begin{itemize}
\item[A)] $6$
\item[B)] $3$
\item[C)] $4.5$
\item[D)] It cannot be summed, since it's defined by a recurrence, not a formula.
\end{itemize}

\ans{A) $6$}
\method{The recurrence $a_{n+1}=\frac12a_n$ with $a_1=3$ generates exactly the GP $3,\frac32,\frac34,\ldots$ (each term is half the last, matching $r=\frac12$). $S_\infty=\dfrac3{1-\frac12}=6$.}
\inv{``Multiply by a constant each time'' is a GP in disguise --- no new machinery needed. \textbf{So what happens if a constant is \emph{added} as well, say $a_{n+1}=\tfrac12a_n+1$? Compute a few terms and see whether they still converge, and to what.} That single extra constant is exactly what Day 4 is built to handle.}

\end{enumerate}

\section*{Top 5 Patterns --- Worksheet \#3}
\begin{enumerate}[leftmargin=2em,itemsep=4pt]
  \item \textbf{Sum to infinity: $S_\infty=\dfrac a{1-r}$, valid only for $|r|<1$.} The central formula of the whole sheet. \seealso{A5, A7, A9, B3, B4}
  \item \textbf{Always check $|r|<1$ before writing down $S_\infty$ at all.} Convergence depends only on $r$, never on $a$. \seealso{A4, A8, B2, B7}
  \item \textbf{Recurring decimals are geometric series in disguise.} $a=$ the repeating block's value (as a fraction of its place value), $r=(\text{base})^{-(\text{block length})}$. \seealso{B8, C5, D2}
  \item \textbf{Day 2's $(1-x)^{-1}=\sum x^k$ \emph{is} the geometric series.} Plug in any $|x|<1$ to evaluate a numeric infinite series instantly. \seealso{B10, C7, C8}
  \item \textbf{Know which formula a question wants: finite $S_n=\dfrac{a(r^n-1)}{r-1}$ or infinite $S_\infty=\dfrac a{1-r}$.} They answer different questions and only one needs a convergence check. \seealso{A3, B1, B9, C2}
\end{enumerate}

\section*{Common Traps --- Worksheet \#3}
\begin{enumerate}[leftmargin=2em,itemsep=4pt]
  \item \textbf{Applying $S_\infty=\frac a{1-r}$ without checking $|r|<1$ first.} The formula still spits out a number even when the series actually diverges --- that number is meaningless. \seealso{B7, C1}
  \item \textbf{Sign errors with negative common ratios inside $1-r$.} $1-(-\frac12)=\frac32$, not $\frac12$ --- expand the double negative carefully. \seealso{A9, B9}
  \item \textbf{Confusing ``converges'' with ``sums to something small.''} A GP with $r$ just under $1$ still converges, but its sum can be enormous. \seealso{C6}
  \item \textbf{Treating a probability question's sample space as only the converging outcomes.} Non-convergent cases stay in the denominator --- they just automatically fail the sum condition, they aren't removed from the count. \seealso{C3}
  \item \textbf{Assuming every recurrence relation needs unfamiliar machinery.} Some recurrences (``multiply by a constant each step'') are secretly a GP, solvable with this sheet's tools already. \seealso{D5}
\end{enumerate}
