\section*{Section A \quad Rapid Recognition}
%═══════════════════════════════════════════════════════════════════════════════
\begin{enumerate}

%── A1 ──────────────────────────────────────────────────────────────────────────
\item Write down row $5$ of Pascal's triangle (the coefficients of $(x+y)^5$).

\ans{$1,5,10,10,5,1$}
\method{Pascal's rule: each entry is the sum of the two above it. Row $4$ is $1,4,6,4,1$; row $5$ adds them pairwise with $1$'s at the ends.}
\inv{Row $n$ has $n+1$ entries and is symmetric: $\binom nk=\binom n{n-k}$. \textbf{So when is the largest entry in a row unique, and when are there two equal largest entries?} Decide it from the parity of $n$, and say what $\binom nk=\binom n{n-k}$ forces at the exact middle.}

%── A2 ──────────────────────────────────────────────────────────────────────────
\item Evaluate $\dbinom{6}{2}$.

\ans{$15$}
\method{$\binom62=\frac{6\times5}{2!}=15$.}
\inv{Equivalently $\binom62=\binom64$ by symmetry --- always compute from the ``short side''. \textbf{Now find every pair $(n,k)$ with $\binom nk=\binom n{k+1}$, i.e.\ two \emph{adjacent} entries equal.} Set the two factorial expressions equal and the condition collapses to a single relation between $n$ and $k$.}

%── A3 ──────────────────────────────────────────────────────────────────────────
\item Write down $\displaystyle\sum_{k=0}^{n}\binom nk$ in closed form.

\ans{$2^n$}
\method{Setting $x=1$ in $(1+x)^n=\sum_k\binom nkx^k$ gives $2^n=\sum_k\binom nk$.}
\inv{Substituting a number for $x$ turns an identity in $x$ into an identity in $n$ --- the most useful trick on the sheet. \textbf{You used $x=1$ to get $2^n$. What do $x=2$ and $x=-2$ give, and what is the general closed form for $\sum_k\binom nkt^k$?} One substitution covers all three.}

%── A4 ──────────────────────────────────────────────────────────────────────────
\item Expand $(1+x)^3$ fully.

\ans{$1+3x+3x^2+x^3$}
\method{Row $3$ of Pascal's triangle is $1,3,3,1$.}
\inv{The coefficients are read straight off Pascal's triangle --- no expanding by hand. \textbf{So, without expanding anything: what is the coefficient of $x^3$ in $(1+x)^3(1+x)^4$?} Combine the powers first. (The slow route --- multiplying the two expansions and collecting terms --- should give the same number; it is worth doing once to see how much work the shortcut saves.)}

%── A5 ──────────────────────────────────────────────────────────────────────────
\item Find the coefficient of $x^2$ in $(1+x)^6$.

\ans{$15$}
\method{$\binom62=15$ (row $6$: $1,6,15,20,15,6,1$).}
\inv{Same computation as A2 --- spotting that saves real time. \textbf{A deeper question: $\binom nk=\frac{n!}{k!(n-k)!}$ involves dividing by $k!(n-k)!$, yet the result is always a whole number. Why must that division always come out exact?} A counting argument answers it instantly; pure algebra struggles.}

%── A6 ──────────────────────────────────────────────────────────────────────────
\item Row $4$ of Pascal's triangle is $1,4,6,4,1$. Is this sequence of five numbers arithmetic? Justify with one check.

\ans{No.}
\method{Consecutive differences are $4-1=3$, $6-4=2$, $4-6=-2$, $1-4=-3$ --- not constant, so the sequence is not arithmetic.}
\inv{Rows are symmetric and unimodal (rise then fall), which rules out any row of more than $2$ entries being arithmetic. \textbf{Can a row ever be \emph{geometric}?} Write the ratio $\binom n{k+1}/\binom nk$ as a formula in $n$ and $k$, then ask what forcing it to be constant in $k$ would require.}

%── A7 ──────────────────────────────────────────────────────────────────────────
\item Simplify $\dbinom n0+\dbinom n1$.

\ans{$1+n$}
\method{$\binom n0=1$, $\binom n1=n$, so the sum is $1+n$.}
\inv{This is the first two terms of the row summing to $2^n$. \textbf{For which $n$ do those first two entries alone account for more than half the row --- that is, $1+n>2^{n-1}$?} Only a couple of small $n$ survive; check them directly rather than solving the inequality.}

%── A8 ──────────────────────────────────────────────────────────────────────────
\item State the value of $\displaystyle\sum_{k=0}^{n}(-1)^k\binom nk$ for $n\geq1$.

\ans{$0$}
\method{Setting $x=-1$ in $(1+x)^n$: $(1-1)^n=0^n=0$ for $n\geq1$, and this equals $\sum_k(-1)^k\binom nk$.}
\inv{The twin of A3: $x=1$ gives the row sum, $x=-1$ gives the alternating sum $0$. \textbf{But the alternating sum is $0$ only for $n\geq1$ --- what does it come to at $n=0$, and why does the argument break there?} Look at what $(1-1)^n$ actually evaluates to when $n=0$.}

%── A9 ──────────────────────────────────────────────────────────────────────────
\item Find the constant term in $(1+x)^5$.

\ans{$1$}
\method{The constant term of $(1+x)^5$ is the $x^0$ term, $\binom50=1$.}
\inv{Every $(1+x)^n$ has constant term $1$, since $\binom n0=1$. \textbf{Now consider $(1+x)^n(1-x)^n$. What is its constant term, and what is its coefficient of $x^2$?} Combine into a single power of $(1-x^2)$ first --- the answer is much cleaner than multiplying two expansions.}

%── A10 ─────────────────────────────────────────────────────────────────────────
\item True or false: row $n$ of Pascal's triangle always has exactly $n+1$ entries.

\ans{True}
\method{Row $n$ has entries $\binom n0,\binom n1,\ldots,\binom nn$ --- that's $n+1$ entries, from $k=0$ to $k=n$ inclusive.}
\inv{A common slip is saying row $n$ has $n$ entries --- count inclusively from $0$. \textbf{So how many entries are there in rows $0$ through $n$ altogether?} You should recognise the answer as a familiar sequence from Day 1 --- and it is worth noticing \emph{why} a triangle of numbers totals a triangular number.}

\end{enumerate}

\section*{Section B \quad Manipulation Drills}
%═══════════════════════════════════════════════════════════════════════════════
\begin{enumerate}

%── B1 ──────────────────────────────────────────────────────────────────────────
\item Prove that $\dbinom n0+\dbinom n1+\cdots+\dbinom nn=2^n$ by substituting a value into the expansion of $(1+x)^n$.

\ans{Proof by substitution $x=1$.}
\method{$(1+x)^n=\sum_{k=0}^n\binom nkx^k$. Setting $x=1$: $(1+1)^n=2^n=\sum_{k=0}^n\binom nk(1)^k=\sum_{k=0}^n\binom nk$. $\square$}
\inv{The cleanest proof of the identity: one substitution, no induction. \textbf{Try to prove the same identity a second way, by induction on $n$ using Pascal's rule $\binom nk=\binom{n-1}{k-1}+\binom{n-1}k$.} Comparing the two proofs shows exactly what the substitution bought you.}

%── B2 ──────────────────────────────────────────────────────────────────────────
\item Row $10$ of Pascal's triangle sums to $2^{10}$. If instead you sum only its first four entries, $\binom{10}{0}+\binom{10}{1}+\binom{10}{2}+\binom{10}{3}$, is the result arithmetic, geometric, or neither, as a rule? Compute the value.

\ans{Neither, in general; the value is $176$.}
\method{$\binom{10}0+\binom{10}1+\binom{10}2+\binom{10}3=1+10+45+120=176$. The four terms themselves ($1,10,45,120$) have differences $9,35,75$ (not constant) and ratios $10,4.5,2.\overline{6}$ (not constant), so the partial sum follows no simple pattern.}
\inv{Individual entries in a row are neither arithmetic nor geometric; only the full row sum ($2^n$) and alternating sum ($0$) are clean. \textbf{What about the sum of just the even-indexed entries, $\binom n0+\binom n2+\binom n4+\cdots$?} Add the $x=1$ and $x=-1$ substitutions together and see what survives.}

%── B3 ──────────────────────────────────────────────────────────────────────────
\item Using $x=-1$ in $(1+x)^n$ ($n\geq1$), what is $\displaystyle\sum_{k=0}^n(-1)^k\binom nk$?
\begin{itemize}
\item[A)] $0$
\item[B)] $1$
\item[C)] $2^n$
\item[D)] $(-1)^n$
\end{itemize}

\ans{A) $0$}
\method{$(1+(-1))^n=0^n=0$ for $n\geq1$.}
\inv{At $n=0$ the sum is $\binom00=1$, not $0$ --- the identity needs $n\geq1$. \textbf{Make a habit of it: for each identity on this sheet, test $n=0$ and $n=1$ before trusting it.} Which of today's identities actually survive $n=0$ intact, and which need the restriction stated?}

%── B4 ──────────────────────────────────────────────────────────────────────────
\item Find the coefficient of $x^3$ in $(2+x)^5$.
\begin{itemize}
\item[A)] $10$
\item[B)] $40$
\item[C)] $80$
\item[D)] $20$
\end{itemize}

\ans{B) $40$}
\method{General term: $\binom5k2^{5-k}x^k$. At $k=3$: $\binom532^2=10\times4=40$.}
\inv{Don't drop the $2^{5-k}$ factor --- answering $\binom53=10$ treats this as $(1+x)^5$. \textbf{Which term of $(2+x)^5$ is the largest when $x=1$, and does the largest coefficient sit at the same place as the largest \emph{term}?} They are not the same question, and that is the point.}

%── B5 ──────────────────────────────────────────────────────────────────────────
\item Find the coefficient of $x^4$ in $(1-2x)^6$.
\begin{itemize}
\item[A)] $60$
\item[B)] $-240$
\item[C)] $240$
\item[D)] $-60$
\end{itemize}

\ans{C) $240$}
\method{General term: $\binom6k(-2x)^k$. At $k=4$: $\binom64(-2)^4=15\times16=240$. The sign is positive since $(-2)^4$ is positive (even power).}
\inv{The trap is $-240$, from forgetting an even power of a negative is positive. \textbf{State the general sign rule: in $(a-b)^n$, exactly which terms are negative?} Give the condition in terms of the index $k$, then check it against this expansion.}

%── B6 ──────────────────────────────────────────────────────────────────────────
\item Find the constant term in the expansion of $\left(x+\dfrac1x\right)^4$.
\begin{itemize}
\item[A)] $4$
\item[B)] $6$
\item[C)] $8$
\item[D)] $1$
\end{itemize}

\ans{B) $6$}
\method{General term: $\binom4kx^{4-k}x^{-k}=\binom4kx^{4-2k}$. Constant term needs $4-2k=0\Rightarrow k=2$: $\binom42=6$.}
\inv{With both positive and negative powers of $x$, set the total exponent to $0$ and solve for $k$ first. \textbf{Now ask when that method returns nothing: for which powers $n$ does $\left(x+\frac1x\right)^n$ have no constant term at all?} Work out the exponent equation in general and see what integrality of $k$ demands.}

%── B7 ──────────────────────────────────────────────────────────────────────────
\item In the expansion of $(1+x)^n$, the coefficient of $x^2$ is $21$ times the coefficient of $x^1$. Find $n$.
\begin{itemize}
\item[A)] $21$
\item[B)] $42$
\item[C)] $43$
\item[D)] $44$
\end{itemize}

\ans{C) $43$}
\method{$\dfrac{\binom n2}{\binom n1}=\dfrac{n-1}{2}=21\Rightarrow n-1=42\Rightarrow n=43$.}
\inv{The ratio trick $\binom n2/\binom n1=\frac{n-1}2$ beats writing both coefficients out. \textbf{Find the general ratio $\binom n{k+1}/\binom nk$ as a formula in $n$ and $k$, then use it to say exactly where a row stops increasing and starts decreasing.} That single formula locates the peak of any row.}

%── B8 ──────────────────────────────────────────────────────────────────────────
\item The coefficients of $x^2$ and $x^3$ in the expansion of $(1+x)^n$ are equal. Find $n$.
\begin{itemize}
\item[A)] $4$
\item[B)] $5$
\item[C)] $6$
\item[D)] Cannot be determined.
\end{itemize}

\ans{B) $5$}
\method{$\binom n2=\binom n3\Rightarrow$ (by the symmetry $\binom nk=\binom n{n-k}$) $2$ and $3$ must be complementary: $n-2=3\Rightarrow n=5$. Check: $\binom52=10=\binom53$. \checkmark}
\inv{This works because $2\neq3$ forces the symmetry route; equal exponents would make any $n$ work and the question ill-posed. \textbf{So state the general rule: given $\binom na=\binom nb$ with $a\neq b$, what is $n$?} One equation, and it should need no case analysis.}

%── B9 ──────────────────────────────────────────────────────────────────────────
\item Find the coefficient of $x^2$ in the expansion of $(1+2x+3x^2)^3$.
\begin{itemize}
\item[A)] $15$
\item[B)] $18$
\item[C)] $21$
\item[D)] $24$
\end{itemize}

\ans{C) $21$}
\method{Trinomial theorem: terms $(1)^a(2x)^b(3x^2)^c$ with $a+b+c=3$, exponent $b+2c=2$. Case $(b,c)=(2,0),a=1$: coefficient $\frac{3!}{1!2!0!}2^2=3\times4=12$. Case $(b,c)=(0,1),a=2$: coefficient $\frac{3!}{2!0!1!}3^1=3\times3=9$. Total: $12+9=21$.}
\inv{Every $(a,b,c)$ with $a+b+c=3$ and $b+2c=2$ must be found systematically --- missed cases are the classic trinomial error. \textbf{Impose a discipline: solve for one variable, list its possible values in order, and stop only when the remaining variables go negative. How many triples are there here, and how do you \emph{know} the list is complete?}}

%── B10 ─────────────────────────────────────────────────────────────────────────
\item The row-sums of Pascal's triangle are $1,2,4,8,16,\ldots$ Is this sequence of row-sums arithmetic, geometric, both, or neither? State the relevant ratio or difference.

\ans{Geometric, common ratio $2$.}
\method{Row sums are $2^0,2^1,2^2,\ldots=1,2,4,8,16,\ldots$ --- each term is exactly double the previous one.}
\inv{In sequence language: the row sums form a GP even though no single row's entries do. \textbf{What is the sum of \emph{all} entries in rows $0$ through $n$, and is that total itself geometric, arithmetic, or neither?} Sum the row sums --- Day 3 will hand you the formula, but you can already guess its shape.}

\end{enumerate}

\section*{Section C \quad Substitution \& Structure}
%═══════════════════════════════════════════════════════════════════════════════
\begin{enumerate}

%── C1 ──────────────────────────────────────────────────────────────────────────
\item Which of the following statements about partial sums of binomial coefficients is correct?
\begin{itemize}
\item[A)] $\binom n0+\binom n1+\cdots+\binom n{k-1}$ always equals $2^n$, for any $k\leq n$.
\item[B)] There is no simple closed form for a general partial sum of binomial coefficients, unlike the full row sum.
\item[C)] Partial sums of binomial coefficients are always arithmetic in $k$.
\item[D)] Partial sums of binomial coefficients are always geometric in $k$.
\end{itemize}

\ans{B) There is no simple closed form for a general partial sum of binomial coefficients, unlike the full row sum.}
\method{The full sum ($k=n+1$ terms) collapses via $x=1$ substitution to $2^n$; a partial sum has no equivalent single substitution trick, and in general must be computed term by term (as in B2's $176$).}
\inv{The exception worth knowing: summing down a \emph{column} across rows does have a closed form --- the hockey-stick identity (C3). \textbf{Before reading C3, guess it: compute $\binom22+\binom32+\binom42+\binom52$ and see which single binomial coefficient it equals.} The pattern is visible from one example.}

%── C2 ──────────────────────────────────────────────────────────────────────────
\item Find the coefficient of $x$ in the expression $(1+x)^0+(1+x)^1+\cdots+(1+x)^{40}$. \textit{\small(after TMUA 2019 Paper 1 Q3)}
\begin{itemize}
\item[A)] $40$
\item[B)] $41$
\item[C)] $820$
\item[D)] $1640$
\item[E)] $1681$
\end{itemize}

\ans{C) $820$}
\method{The coefficient of $x$ in $(1+x)^k$ is $\binom k1=k$. Summing over $k=0$ to $40$: $\sum_{k=0}^{40}k=\frac{40\times41}2=820$.}
\inv{Day 1's AP sum formula, now extracting information from a sum of binomial expansions --- the two toolkits meet here. \textbf{Try the real TMUA version (2019 Paper 1 Q3), which sums to $k=80$ instead of $40$.} Predict the answer from the structure before computing it, then check you get $\frac{80\times81}2=3240$.}

%── C3 ──────────────────────────────────────────────────────────────────────────
\item Find, in terms of $n$, the coefficient of $x^2$ in the expression $(1+x)^0+(1+x)^1+\cdots+(1+x)^n$. \textit{\small(after TMUA 2019 Paper 1 Q3)}
\begin{itemize}
\item[A)] $\dbinom n3$
\item[B)] $\dbinom{n+1}3$
\item[C)] $\dbinom{n+1}2$
\item[D)] $2^n-n-1$
\end{itemize}

\ans{B) $\dbinom{n+1}3$}
\method{The coefficient of $x^2$ in $(1+x)^k$ is $\binom k2$ (zero for $k<2$). Summing $\sum_{k=2}^n\binom k2=\binom{n+1}3$ is the hockey-stick identity: summing a column of Pascal's triangle from its start lands one row down and one column right.}
\inv{\textbf{Prove the hockey-stick identity by induction on $n$ using Pascal's rule, then explain why C2 is the same identity one column to the left} (with $\binom k1=k$ in place of $\binom k2$). \textbf{Finally: what does the column one step further right, $\binom k3$, sum to?} Extend the pattern and verify it on a small case.}

%── C4 ──────────────────────────────────────────────────────────────────────────
\item In the expansion of $(1+x+x^2)^4$, what is the coefficient of $x^3$?
\begin{itemize}
\item[A)] $12$
\item[B)] $16$
\item[C)] $20$
\item[D)] $24$
\end{itemize}

\ans{B) $16$}
\method{Trinomial theorem on $(1+x+x^2)^4$: terms $(1)^a(x)^b(x^2)^c$, $a+b+c=4$, exponent $b+2c=3$. Case $(b,c)=(3,0),a=1$: $\frac{4!}{1!3!0!}=4$. Case $(b,c)=(1,1),a=2$: $\frac{4!}{2!1!1!}=12$. Total: $4+12=16$.}
\inv{Same systematic case-finding as B9 at a larger power --- list every $(b,c)$ pair before computing any coefficient. \textbf{How many terms does the full expansion of $(1+x+x^2)^n$ have before collecting like terms, and how many distinct powers of $x$ can appear after collecting?} The second count is far smaller, and knowing it in advance tells you how much casework to expect.}

%── C5 ──────────────────────────────────────────────────────────────────────────
\item Which of the following is the expansion of $(1-x)^{-1}$ as an infinite series in $x$?
\begin{itemize}
\item[A)] $1-x+x^2-x^3+\cdots$
\item[B)] $1+x+x^2+x^3+\cdots$
\item[C)] $1-2x+3x^2-4x^3+\cdots$
\item[D)] It cannot be expanded as a series.
\end{itemize}

\ans{B) $1+x+x^2+x^3+\cdots$}
\method{Long division, or the generalized binomial series $(1-x)^{-1}=\sum_{k=0}^\infty\binom{-1}kx^k$ with $(-x)^k$ sign conventions, both give every coefficient equal to $1$.}
\inv{This is the geometric series with $a=1,r=x$ in disguise, and the whole of Day 3 is built on it. \textbf{Verify it by multiplying out $(1-x)(1+x+x^2+\cdots)$ and watching it telescope to $1$ --- then say precisely which values of $x$ make that manipulation legitimate.} The algebra looks valid for every $x$; it is not.}

%── C6 ──────────────────────────────────────────────────────────────────────────
\item The generalized binomial expansion gives $(1-x)^{-2}=\sum_{k=0}^\infty(k+1)x^k$. Find the coefficient of $x^3$.
\begin{itemize}
\item[A)] $3$
\item[B)] $4$
\item[C)] $6$
\item[D)] $10$
\end{itemize}

\ans{B) $4$}
\method{At $k=3$: coefficient is $k+1=4$.}
\inv{The coefficients $k+1$ form an AP with $d=1$: differentiating $(1-x)^{-1}$ turned a constant coefficient sequence into a linear one. \textbf{So predict, before computing: what sort of sequence are the coefficients of $(1-x)^{-3}$, and what is the coefficient of $x^k$?} Differentiate once more and check your prediction against C8.}

%── C7 ──────────────────────────────────────────────────────────────────────────
\item Which of the following is FALSE, for a positive integer $n$?
\begin{itemize}
\item[A)] $\displaystyle\sum_{k=0}^n\binom nk=2^n$
\item[B)] $\displaystyle\sum_{k=0}^n(-1)^k\binom nk=0$
\item[C)] $\dbinom nk=\dbinom n{n-k}$ for all $0\leq k\leq n$
\item[D)] $\displaystyle\sum_{k=0}^n k\binom nk=2^n$
\end{itemize}

\ans{D) $\displaystyle\sum_{k=0}^n k\binom nk=2^n$}
\method{The true identity is $\sum_{k=0}^nk\binom nk=n\cdot2^{n-1}$ (proved in D3), not $2^n$. Check $n=3$: true value is $3\times4=12$, but D claims $2^3=8\neq12$.}
\inv{Three genuine identities beside one plausible fabrication --- the defence is always to test a small case. \textbf{Take the false option and find the smallest $n$ at which it fails.} Then make it a habit: for any half-remembered binomial identity, which single value of $n$ is fastest to test, and which is most likely to be misleading?}

%── C8 ──────────────────────────────────────────────────────────────────────────
\item The infinite series $1+x+x^2+x^3+\cdots$ equals $(1-x)^{-1}$. Using $(1-x)^{-3}=\sum_{k=0}^\infty\binom{k+2}{2}x^k$, find the coefficient of $x^5$ in $(1-x)^{-3}$.
\begin{itemize}
\item[A)] $\dbinom52$
\item[B)] $\dbinom62$
\item[C)] $\dbinom72$
\item[D)] $\dbinom82$
\end{itemize}

\ans{C) $\dbinom72$}
\method{$(1-x)^{-3}=\sum_{k=0}^\infty\binom{k+2}2x^k$. At $k=5$: $\binom{5+2}2=\binom72=21$.}
\inv{Across $(1-x)^{-1},(1-x)^{-2},(1-x)^{-3}$ the coefficient of $x^k$ is $\binom k0,\binom{k+1}1,\binom{k+2}2$ --- ``stars and bars in series form''. \textbf{Write down the coefficient of $x^k$ in $(1-x)^{-m}$ for general $m$, then check your formula reproduces all three cases above.} Test $m=1$ especially --- it is the one most likely to expose an off-by-one.}

\end{enumerate}

\section*{Section D \quad Challenge}
%═══════════════════════════════════════════════════════════════════════════════
\begin{enumerate}

%── D1 ──────────────────────────────────────────────────────────────────────────
\item Find the coefficient of $x^2y^3$ in the expansion of $(1+x+y^2)^5$. \textit{\small(after TMUA 2020 Paper 1 Q13)}
\begin{itemize}
\item[A)] $0$
\item[B)] $10$
\item[C)] $20$
\item[D)] $30$
\item[E)] $60$
\end{itemize}

\ans{A) $0$}
\method{General term of $(1+x+y^2)^5$: $(1)^a(x)^b(y^2)^c$ with $a+b+c=5$, contributing $x^by^{2c}$. The power of $y$ is always $2c$ --- an even number. Since $3$ is odd, there is no valid $(a,b,c)$ giving $x^2y^3$, so its coefficient is $0$.}
\inv{The reward here is recognising an \emph{impossible} target instantly instead of grinding through cases to find them all empty. \textbf{Make the parity argument general: in $(1+x+y^2)^n$, exactly which coefficients of $x^ay^b$ are forced to be zero?} State the rule in one line, then use it to answer the real TMUA version (2020 Paper 1 Q13), which asks for the achievable $x^2y^4$.}

%── D2 ──────────────────────────────────────────────────────────────────────────
\item The coefficient of $x^2$ in $(1+x+2x^2)^3$ equals the coefficient of $x^2$ in $(1-ax)^3$. Find the possible value(s) of $a$. \textit{\small(after TMUA Practice Paper 1 Q19)}
\begin{itemize}
\item[A)] $a=\pm\sqrt3$
\item[B)] $a=\pm3$
\item[C)] $a=\sqrt3$ only
\item[D)] $a=3$ only
\end{itemize}

\ans{A) $a=\pm\sqrt3$}
\method{Coefficient of $x^2$ in $(1+x+2x^2)^3$: cases $(b,c)=(2,0),a{=}1$ gives $\frac{3!}{1!2!0!}1^2=3$; $(b,c)=(0,1),a{=}2$ gives $\frac{3!}{2!0!1!}2^1=6$. Total $3+6=9$. Coefficient of $x^2$ in $(1-ax)^3$: $\binom32(-a)^2=3a^2$. Setting equal: $3a^2=9\Rightarrow a^2=3\Rightarrow a=\pm\sqrt3$.}
\inv{\textbf{Work through the real version by hand (Practice Paper 1 Q19): match the coefficient of $x^3$ in $(1+2x+3x^2)^6$ against \emph{twice} the coefficient of $x^4$ in $(1-ax^2)^5$, and confirm $a=\pm\sqrt{17}$.} Then explain why $a$ comes out as $\pm$ a surd here while the sheet's version gives $\pm\sqrt3$ --- what feature of the equation guarantees two solutions rather than one?}

%── D3 ──────────────────────────────────────────────────────────────────────────
\item Prove that $\displaystyle\sum_{k=0}^n k\binom nk=n\cdot2^{n-1}$ for every positive integer $n$, by differentiating the binomial expansion of $(1+x)^n$.

\ans{Proof by differentiating $(1+x)^n$.}
\method{$(1+x)^n=\sum_{k=0}^n\binom nkx^k$. Differentiating both sides with respect to $x$: $n(1+x)^{n-1}=\sum_{k=0}^nk\binom nkx^{k-1}$ (the $k=0$ term vanishes since it has no $x$). Setting $x=1$: $n\cdot2^{n-1}=\sum_{k=0}^nk\binom nk$. $\square$}
\inv{The standard move for weighted binomial sums: differentiate first, substitute second. \textbf{Now evaluate $\sum_{k=0}^nk^2\binom nk$ the same way} --- differentiate $x\cdot n(1+x)^{n-1}$ once more before substituting. \textbf{Check your answer at $n=2$ by hand}, where the sum is short enough to verify term by term.}

%── D4 ──────────────────────────────────────────────────────────────────────────
\item Expanding $(1+x)^n(1+x)^m$ two different ways --- first by combining the factors as $(1+x)^{n+m}$, then by multiplying the two separate expansions together and comparing the coefficient of $x^k$ on both sides --- produces which identity?
\begin{itemize}
\item[A)] Vandermonde's identity: $\displaystyle\sum_{i=0}^k\binom ni\binom m{k-i}=\binom{n+m}k$.
\item[B)] $\dbinom nk+\dbinom mk=\dbinom{n+m}k$.
\item[C)] $\dbinom nk\dbinom mk=\dbinom{n+m}k$.
\item[D)] No identity results --- the two sides are unrelated.
\end{itemize}

\ans{A) Vandermonde's identity.}
\method{$(1+x)^n(1+x)^m=\left(\sum_i\binom nix^i\right)\left(\sum_j\binom mjx^j\right)$. The coefficient of $x^k$ on the right is $\sum_{i=0}^k\binom ni\binom m{k-i}$ (pairing every way $i+j=k$). Since the left side is also $(1+x)^{n+m}$, whose coefficient of $x^k$ is $\binom{n+m}k$, the two must be equal.}
\inv{Vandermonde also reads combinatorially: choose $k$ people from $n$ men and $m$ women, split by how many men. \textbf{Set $m=n$ and $k=n$ to get a striking special case --- what single binomial coefficient does $\sum_k\binom nk^2$ equal?} Use the symmetry $\binom nk=\binom n{n-k}$ to see why the squares appear, then verify at $n=3$.}

%── D5 ──────────────────────────────────────────────────────────────────────────
\item The coefficients of $(1-x)^{-1}$ are all $1$ (constant in $k$). The coefficients of $(1-x)^{-2}$, namely $(k+1)$, form an arithmetic sequence in $k$. What kind of sequence, as a function of $k$, do the coefficients of $(1-x)^{-3}=\sum_{k=0}^\infty\binom{k+2}2x^k$ form?
\begin{itemize}
\item[A)] Arithmetic.
\item[B)] Geometric.
\item[C)] Quadratic in $k$ (neither arithmetic nor geometric).
\item[D)] Constant.
\end{itemize}

\ans{C) Quadratic in $k$ (neither arithmetic nor geometric).}
\method{$\binom{k+2}2=\dfrac{(k+2)(k+1)}2=\dfrac12k^2+\dfrac32k+1$, a genuine quadratic in $k$ (nonzero $k^2$ coefficient).}
\inv{Day 1's ``$S_n$ is quadratic'' pattern in disguise: the coefficients of $(1-x)^{-1},(1-x)^{-2},(1-x)^{-3}$ are degree $0,1,2$ in $k$, each inverse power raising the degree by one. \textbf{Is that a coincidence, or the same fact twice?} Consider what multiplying a series by $(1-x)^{-1}$ does to its coefficients --- and compare that operation with taking partial sums of a sequence.}

\end{enumerate}

\section*{Top 5 Patterns --- Worksheet \#2}
\begin{enumerate}[leftmargin=2em,itemsep=4pt]
  \item \textbf{Row sum: $\sum_k\binom nk=2^n$, from setting $x=1$ in $(1+x)^n$.} The single most useful substitution in the whole sheet. \seealso{A3, B1, C2}
  \item \textbf{Alternating sum: $\sum_k(-1)^k\binom nk=0$ for $n\geq1$, from setting $x=-1$.} The ``twin'' of the row-sum trick. \seealso{A8, B3}
  \item \textbf{The generalized binomial series bridges straight into geometric series.} $(1-x)^{-m}=\sum_{k=0}^\infty\binom{k+m-1}{m-1}x^k$ --- at $m=1$ this is exactly the geometric series. \seealso{C5, C6, C8, D5}
  \item \textbf{Trinomial/multinomial coefficient extraction: find every $(a,b,c)$ with $a{+}b{+}c=$ power and the right exponent.} Missing a case is the single most common error. \seealso{B9, C4, D1, D2}
  \item \textbf{Differentiate $(1+x)^n$ term-by-term to extract weighted sums like $\sum k\binom nk$.} Substitute $x=1$ after differentiating, not before. \seealso{C7, D3}
\end{enumerate}

\section*{Common Traps --- Worksheet \#2}
\begin{enumerate}[leftmargin=2em,itemsep=4pt]
  \item \textbf{Confusing a partial binomial sum with the full row sum.} Only the complete sum $k=0$ to $n$ has the clean $2^n$ closed form --- a partial sum in general has no shortcut. \seealso{C1}
  \item \textbf{Assuming every target power is achievable in a trinomial expansion.} $(1+x+y^2)^n$ can only ever produce \emph{even} powers of $y$ --- an odd target power isn't a computation to grind through, it's genuinely zero. \seealso{D1}
  \item \textbf{Off-by-one errors between $\binom{n+1}3$ and $\binom n3$ in hockey-stick-style partial sums.} Summing a column of Pascal's triangle lands one row \emph{down}, not on the same row. \seealso{C3}
  \item \textbf{Assuming binomial-series coefficients stay linear (arithmetic) in $k$ for every power of $(1-x)^{-m}$.} Only $m=2$ gives an AP --- higher $m$ gives higher-degree polynomials in $k$. \seealso{D5}
  \item \textbf{Forgetting the multinomial coefficient needs all three factorials, $\frac{n!}{p!q!r!}$, not just a single $\binom nk$.} A trinomial expansion is not two nested binomial expansions in disguise. \seealso{B9, C4}
\end{enumerate}
