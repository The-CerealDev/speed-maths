% ── Speed Sheet overrides (loaded only by tools/build_speed_sheets.py) ─────────
% Input at \begin{document} when an answers file is compiled as a Speed Sheet;
% the build defines \SpeedAnsURL (that sheet's published answers PDF) and
% \SpeedSheetTitle. Nothing here may \usepackage: the preamble is already shut,
% so everything leans on what shared/preamble.tex loaded (graphicx, hyperref,
% xcolor, titlesec).
\makeatletter

% Working space under each question; Section A-D headings change it.
\newlength{\SpeedSpace}\setlength{\SpeedSpace}{3cm}

% Start a new page now if less than #1 is left on this one (\Needspace, inlined).
% No "is anything left" guard: a page filled exactly to the bottom must break
% too, or the next question lands there with no room. An empty page has
% \pagegoal = \maxdimen, so this never breaks needlessly.
\newcommand{\Speed@need}[1]{\par\penalty-100\begingroup
  \dimen@=#1\relax\dimen@ii=\pagegoal\advance\dimen@ii-\pagetotal
  \ifdim\dimen@>\dimen@ii\break\fi\endgroup}

% A question never splits from its space. The build runs pdflatex more than
% once: each pass writes every question's measured height to the .aux, and the
% next pass reserves that height plus the section's space before starting the
% question. Until a height is known, 3cm stands in for it.
% Only top-level items outside any box are questions: MCQ options and sub-parts
% sit deeper, or inside a minipage (\ifinner), and are left alone.
\newlength{\Speed@start}
\newcount\Speed@startpage
\newcommand{\Speed@height}[1]{%
  \ifcsname Speed@h@#1\endcsname\csname Speed@h@#1\endcsname\else3cm\fi}
\let\Speed@item\item
\renewcommand{\item}{\ifnum\@listdepth=\@ne\ifinner\else
  \Speed@need{\dimexpr\SpeedSpace+\Speed@height{\the\numexpr\c@Item+1\relax}+12mm\relax}%
  \global\Speed@start=\pagetotal\global\Speed@startpage=\c@page
  \fi\fi\Speed@item}
\newcommand{\Speed@record}{%
  \ifnum\Speed@startpage=\c@page\ifdim\pagetotal>\Speed@start
    \immediate\write\@auxout{\string\expandafter\string\gdef\string\csname\space
      Speed@h@\the\c@Item\string\endcsname{\the\dimexpr\pagetotal-\Speed@start\relax}}%
  \fi\fi}

% "Section A ..." to "Section D ..." set the space and keep the heading with a
% question. Any other starred section (Top 5 Patterns, Common Traps) would give
% answers away, so the sheet ends there.
\newcommand{\Speed@try}[3]{\in@{Section #1}{#3}\ifin@\setlength{\SpeedSpace}{#2}\fi}
\let\Speed@section\section
\renewcommand{\section}{\@ifstar\Speed@ssection\Speed@section}
\newcommand{\Speed@ssection}[1]{%
  \in@{Section }{#1}%
  \ifin@
    \Speed@try{A}{3cm}{#1}\Speed@try{B}{4.5cm}{#1}%
    \Speed@try{C}{6cm}{#1}\Speed@try{D}{7cm}{#1}%
    % the heading, its instruction line, and the whole of the first question
    \Speed@need{\dimexpr3cm+\SpeedSpace+\Speed@height{\the\numexpr\c@Item+1\relax}+12mm\relax}%
    \Speed@section*{#1}%
  \else
    \end{document}%
  \fi}

\let\Speed@title\SpeedTitleBlock
\renewcommand{\SpeedTitleBlock}[2]{\Speed@title{\SpeedSheetTitle}{#2}}

% The answer: upside down, 4pt, faint, its right edge 16mm into the 22mm margin.
% The whole box links to this question's worked answer; it is at least 15mm x 5mm
% so a finger can hit it, though the text inside stays tiny.
\newsavebox{\Speed@ansbox}
\newsavebox{\Speed@linkbox}
\newlength{\Speed@w}
% A tall question (diagram, MCQ options) can leave less than \SpeedSpace on its
% page. The space then shrinks to what is left, so the answer stays at the foot
% of its own question instead of being pushed alone onto the next page.
\newlength{\Speed@room}
\newcommand{\Speed@space}{\par\penalty\@M\Speed@record
  \setlength{\Speed@room}{\dimexpr\pagegoal-\pagetotal-12mm\relax}%
  \ifdim\Speed@room<\SpeedSpace
    \ifdim\Speed@room>\z@\vspace{\Speed@room}\fi\nopagebreak
  \else\vspace{\SpeedSpace}\fi}
\renewcommand{\ans}[1]{\Speed@space%
  \sbox{\Speed@ansbox}{\rotatebox{180}{\color{black!22}\fontsize{4}{5}\selectfont Ans: #1}}%
  \setlength{\Speed@w}{\wd\Speed@ansbox}%
  \ifdim\Speed@w<15mm\setlength{\Speed@w}{15mm}\fi
  \sbox{\Speed@linkbox}{\href{\SpeedAnsURL\#nameddest=Item.\arabic{Item}}{%
    \makebox[\Speed@w][r]{\rule{0pt}{5mm}\usebox{\Speed@ansbox}}}}%
  \noindent\hfill\rlap{\hspace{\dimexpr16mm-\wd\Speed@linkbox\relax}\usebox{\Speed@linkbox}}%
  \par\medskip}
\renewcommand{\method}[1]{}
\renewcommand{\inv}[1]{}

\makeatother
